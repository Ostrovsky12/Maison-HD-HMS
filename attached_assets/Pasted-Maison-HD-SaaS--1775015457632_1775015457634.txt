// server/routes/admin/organizations.ts

import { Router } from 'express';
import { db } from '../../db';
import { organizations, organizationUsers, users } from '../../db/schema';
import { hash } from 'bcrypt';
import { sendInvitationEmail } from '../../services/email';

const router = Router();

// POST /api/admin/organizations
router.post('/organizations', async (req, res) => {
  const { 
    name, 
    slug, 
    email, 
    phone, 
    address,
    ownerEmail,
    ownerName,
    ownerPassword,
    subscriptionTier = 'pro',
    trialDays = 30,
  } = req.body;
  
  // 1. Створюємо організацію
  const [organization] = await db.insert(organizations).values({
    name,
    slug,
    email,
    phone,
    address,
    subscriptionTier,
    trialEndsAt: new Date(Date.now() + trialDays * 24 * 60 * 60 * 1000),
    subscriptionStatus: 'trial',
  }).returning();
  
  // 2. Створюємо користувача-власника
  const hashedPassword = await hash(ownerPassword, 10);
  const [owner] = await db.insert(users).values({
    email: ownerEmail,
    fullName: ownerName,
    passwordHash: hashedPassword,
  }).returning();
  
  // 3. Прив'язуємо власника до організації
  await db.insert(organizationUsers).values({
    organizationId: organization.id,
    userId: owner.id,
    role: 'owner',
    permissions: ['*'], // всі права
    invitedAt: new Date(),
    acceptedAt: new Date(),
  });
  
  // 4. Відправляємо привітальний email
  await sendInvitationEmail(ownerEmail, organization.name, {
    loginUrl: `https://app.maison-hd.com/${organization.slug}/login`,
    email: ownerEmail,
    password: ownerPassword,
  });
  
  res.json({
    success: true,
    data: {
      organization,
      owner: { id: owner.id, email: owner.email },
      loginUrl: `https://app.maison-hd.com/${organization.slug}/login`,
    },
  });
});

// GET /api/admin/organizations
router.get('/organizations', async (req, res) => {
  const orgs = await db.select()
    .from(organizations)
    .orderBy(organizations.createdAt, 'desc');
  
  res.json({ success: true, data: orgs });
});

export default router;

import { useState } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { api } from '@/lib/api';
import { toast } from 'sonner';

export function OnboardingWizard({ organizationId }: { organizationId: number }) {
  const [step, setStep] = useState(1);
  
  const { data: property } = useQuery({
    queryKey: ['property', organizationId],
    queryFn: () => api.get(`/properties?organizationId=${organizationId}`),
  });
  
  const hasProperty = property?.data?.length > 0;
  
  // Якщо заклад ще не налаштований — показуємо візард
  if (!hasProperty) {
    return (
      <div className="max-w-3xl mx-auto py-10">
        <div className="text-center mb-8">
          <h1 className="text-2xl font-bold">Вітаємо в Maison HD!</h1>

            organizationId={organizationId}

          />
        )}
        
        {step === 2 && (
          <SetupRoomsForm 
            organizationId={organizationId} 
            onComplete={() => setStep(3)} 
          />
        )}
        
        {step === 3 && (
          <InviteStaffForm 
            organizationId={organizationId}

        {step === 4 && (

function SetupRoomsForm({ organizationId, onComplete }: { organizationId: number; onComplete: () => void }) {

    onSuccess: () => {
      toast.success('Запрошення надіслано');
      onComplete();
    },
  });
  
  return (
    <div className="bg-white rounded-xl shadow-sm p-6">
      <h2 className="text-lg font-semibold mb-4">Запросити співробітників</h2>
      <p className="text-sm text-gray-500 mb-4">
        Співробітники отримають email з посиланням для реєстрації
      </p>
      
      {invites.map((invite, idx) => (
        <div key={idx} className="grid grid-cols-3 gap-3 mb-3">
          <input 
            placeholder="Email"
            className="border rounded-lg px-3 py-2"
            value={invite.email}
            onChange={(e) => {
              const newInvites = [...invites];
              newInvites[idx].email = e.target.value;
              setInvites(newInvites);
            }}
          />
          <input 
            placeholder="ПІБ"
            className="border rounded-lg px-3 py-2"
            value={invite.name}
            onChange={(e) => {
              const newInvites = [...invites];
              newInvites[idx].name = e.target.value;
              setInvites(newInvites);
            }}
          />
          <select 
            className="border rounded-lg px-3 py-2"
            value={invite.role}
            onChange={(e) => {
              const newInvites = [...invites];
              newInvites[idx].role = e.target.value;
              setInvites(newInvites);
            }}
          >
            <option value="reception">Адміністратор рецепції</option>
            <option value="housekeeper">Покоївка</option>
            <option value="accountant">Бухгалтер</option>
          </select>
        </div>
      ))}
      
      <Button 
        variant="outline" 
        size="sm" 
        className="mt-2"
        onClick={() => setInvites([...invites, { email: '', role: 'reception', name: '' }])}
      >
        + Додати ще
      </Button>
      
      <div className="flex justify-between pt-6">
        <Button variant="outline" onClick={onComplete}>Пропустити</Button>
        <Button onClick={() => mutation.mutate()}>
          {mutation.isPending ? 'Надсилання...' : 'Надіслати запрошення →'}
        </Button>
      </div>
    </div>
  );
}

// src/components/reception/BookingCalendar.tsx

import { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { api } from '@/lib/api';

export function BookingCalendar({ propertyId }: { propertyId: number }) {
  const [currentDate, setCurrentDate] = useState(new Date());
  
  const { data: calendar } = useQuery({
    queryKey: ['calendar', propertyId, currentDate],
    queryFn: () => api.get(`/bookings/calendar?propertyId=${propertyId}&month=${currentDate.getMonth()+1}&year=${currentDate.getFullYear()}`),
  });
  
  const weekdays = ['ПН', 'ВТ', 'СР', 'ЧТ', 'ПТ', 'СБ', 'НД'];
  const daysInMonth = new Date(currentDate.getFullYear(), currentDate.getMonth() + 1, 0).getDate();
  const firstDay = new Date(currentDate.getFullYear(), currentDate.getMonth(), 1).getDay() || 7;
  
  return (
    <div className="bg-white rounded-xl shadow-sm p-6">
      <div className="flex justify-between items-center mb-6">
        <h2 className="text-lg font-semibold">
          {currentDate.toLocaleString('uk-UA', { month: 'long', year: 'numeric' })}
        </h2>
        <div className="flex space-x-2">
          <button 
            onClick={() => setCurrentDate(new Date(currentDate.setMonth(currentDate.getMonth() - 1)))}
            className="p-2 hover:bg-gray-100 rounded-lg"
          >
            ←
          </button>
          <button 
            onClick={() => setCurrentDate(new Date())}
            className="px-3 py-1 text-sm border rounded-lg hover:bg-gray-50"
          >
            Сьогодні
          </button>
          <button 
            onClick={() => setCurrentDate(new Date(currentDate.setMonth(currentDate.getMonth() + 1)))}
            className="p-2 hover:bg-gray-100 rounded-lg"
          >
            →
          </button>
        </div>
      </div>
      
      <div className="overflow-x-auto">
        <table className="w-full border-collapse">
          <thead>
            <tr>
              {weekdays.map(day => (
                <th key={day} className="p-3 text-center text-sm font-medium text-gray-500">
                  {day}
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {Array.from({ length: Math.ceil((daysInMonth + firstDay - 1) / 7) }).map((_, weekIdx) => (
              <tr key={weekIdx}>
                {Array.from({ length: 7 }).map((_, dayIdx) => {
                  const dayNumber = weekIdx * 7 + dayIdx - firstDay + 2;
                  const isValid = dayNumber >= 1 && dayNumber <= daysInMonth;
                  const date = `${currentDate.getFullYear()}-${String(currentDate.getMonth()+1).padStart(2,'0')}-${String(dayNumber).padStart(2,'0')}`;
                  const bookings = calendar?.data?.bookings?.[date] || [];
                  const occupancy = calendar?.data?.occupancy?.[date] || 0;
                  
                  return (
                    <td 
                      key={dayIdx}
                      className={`
                        p-2 border text-center align-top cursor-pointer hover:bg-indigo-50
                        ${!isValid ? 'bg-gray-50' : ''}
                        ${occupancy >= 90 ? 'bg-red-50' : occupancy >= 50 ? 'bg-yellow-50' : 'bg-green-50'}
                      `}
                      onClick={() => isValid && console.log('Select date', date)}
                    >
                      {isValid && (
                        <>
                          <div className="font-medium">{dayNumber}</div>
                          <div className="text-xs text-gray-500 mt-1">
                            {bookings.length > 0 && (
                              <span className="block">{bookings.length} бронь</span>
                            )}
                            {occupancy > 0 && (
                              <span className="block text-xs font-medium">
                                {Math.round(occupancy)}%
                              </span>
                            )}
                          </div>
                        </>
                      )}
                    </td>
                  );
                })}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}
