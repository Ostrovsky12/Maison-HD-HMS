// src/components/booking/Chessboard.tsx (оновлений)

import { useState, useRef, useEffect, useCallback } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { api } from '@/lib/api';
import { toast } from 'sonner';
import {
  ChevronLeft,
  ChevronRight,
  Calendar,
  Filter,
  Download,
  Settings,
  Plus,
  Bed,
  Users,
  Clock,
  Eye,
  MoreVertical,
  X,
  Tag,
  Home,
  Building,
  Trees,
  Hotel,
  Sparkles,
} from 'lucide-react';

// Іконки для категорій
const categoryIcons: Record<string, any> = {
  standard: Bed,
  comfort: Hotel,
  suite: Sparkles,
  luxury: Sparkles,
  cottage: Trees,
  apartment: Building,
  family: Users,
  default: Home,
};

// Кольори для категорій
const categoryColors: Record<string, string> = {
  standard: 'bg-blue-50 text-blue-700 border-blue-200',
  comfort: 'bg-green-50 text-green-700 border-green-200',
  suite: 'bg-purple-50 text-purple-700 border-purple-200',
  luxury: 'bg-amber-50 text-amber-700 border-amber-200',
  cottage: 'bg-emerald-50 text-emerald-700 border-emerald-200',
  apartment: 'bg-indigo-50 text-indigo-700 border-indigo-200',
  family: 'bg-pink-50 text-pink-700 border-pink-200',
  default: 'bg-gray-50 text-gray-700 border-gray-200',
};

interface Category {
  id: number;
  name: string;
  code: string;
  color: string;
  icon: string;
  count: number;
  basePrice: number;
}

interface Room {
  id: number;
  number: string;
  name: string;
  categoryId: number;
  category: Category;
  capacity: number;
  floor: number;
  status: 'clean' | 'dirty' | 'repair' | 'occupied';
}

interface ChessboardProps {
  propertyId: number;
  startDate?: Date;
}

export function Chessboard({ propertyId, startDate = new Date() }: ChessboardProps) {
  // Стан
  const [currentDate, setCurrentDate] = useState(startDate);
  const [viewMode, setViewMode] = useState<'day' | 'week' | 'month'>('month');
  const [selectedRooms, setSelectedRooms] = useState<number[]>([]);
  const [selectionRange, setSelectionRange] = useState<{ start: string | null; end: string | null }>({ start: null, end: null });
  const [isSelecting, setIsSelecting] = useState(false);
  const [tooltip, setTooltip] = useState<{ x: number; y: number; text: string; visible: boolean }>({ x: 0, y: 0, text: '', visible: false });
  const [showCreateBooking, setShowCreateBooking] = useState(false);
  const [hoveredCell, setHoveredCell] = useState<{ roomId: number; date: string } | null>(null);
  
  // Фільтри
  const [selectedCategories, setSelectedCategories] = useState<number[]>([]);
  const [showFilters, setShowFilters] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [showOnlyAvailable, setShowOnlyAvailable] = useState(false);
  const [sortBy, setSortBy] = useState<'number' | 'floor' | 'capacity' | 'price'>('number');
  
  const chessboardRef = useRef<HTMLDivElement>(null);
  const cellRefs = useRef<Map<string, HTMLDivElement>>(new Map());
  
  // Отримання категорій
  const { data: categories, isLoading: categoriesLoading } = useQuery({
    queryKey: ['categories', propertyId],
    queryFn: () => api.get(`/properties/${propertyId}/categories`),
  });
  
  // Отримання номерів з фільтрами
  const { data: rooms, isLoading: roomsLoading } = useQuery({
    queryKey: ['rooms', propertyId, selectedCategories, searchQuery, sortBy],
    queryFn: () => api.get(`/properties/${propertyId}/rooms`, {
      params: {
        categories: selectedCategories.join(','),
        search: searchQuery,
        sortBy,
        onlyAvailable: showOnlyAvailable,
      },
    }),
  });
  
  // Отримання бронювань
  const { data: bookings, refetch: refetchBookings } = useQuery({
    queryKey: ['bookings', propertyId, currentDate, viewMode],
    queryFn: () => api.get(`/properties/${propertyId}/bookings/calendar`, {
      params: {
        startDate: currentDate.toISOString().split('T')[0],
        days: viewMode === 'month' ? 35 : viewMode === 'week' ? 7 : 1,
      },
    }),
  });
  
  // Фільтрування номерів за категоріями
  const filteredRooms = rooms?.data?.filter((room: Room) => {
    if (selectedCategories.length === 0) return true;
    return selectedCategories.includes(room.categoryId);
  }) || [];
  
  // Генерація днів
  const generateDays = useCallback(() => {
    const days = [];
    const start = new Date(currentDate);
    
    if (viewMode === 'month') {
      start.setDate(1);
      const firstDayOfMonth = start.getDay() || 7;
      start.setDate(start.getDate() - (firstDayOfMonth - 1));
      
      for (let i = 0; i < 35; i++) {
        const date = new Date(start);
        date.setDate(start.getDate() + i);
        days.push(date);
      }
    } else if (viewMode === 'week') {
      const dayOfWeek = start.getDay() || 7;
      start.setDate(start.getDate() - (dayOfWeek - 1));
      
      for (let i = 0; i < 7; i++) {
        const date = new Date(start);
        date.setDate(start.getDate() + i);
        days.push(date);
      }
    } else {
      days.push(new Date(currentDate));
    }
    
    return days;
  }, [currentDate, viewMode]);
  
  const days = generateDays();
  
  // Отримання бронювання для клітинки
  const getBookingForCell = (roomId: number, date: Date): any => {
    if (!bookings?.data) return null;
    const dateStr = date.toISOString().split('T')[0];
    return bookings.data.find(
      (b: any) => b.roomId === roomId && b.checkIn <= dateStr && b.checkOut > dateStr
    ) || null;
  };
  
  // Визначення статусу клітинки
  const getCellStatus = (room: Room, date: Date, booking: any) => {
    if (booking) {
      if (booking.status === 'checked_in') return 'occupied';
      if (booking.status === 'confirmed') return 'booked';
      return 'blocked';
    }
    
    if (room.status === 'dirty') return 'dirty';
    if (room.status === 'repair') return 'repair';
    return 'free';
  };
  
  // Перевірка чи дата в діапазоні виділення
  const isInSelectionRange = (roomId: number, date: Date): boolean => {
    if (!selectionRange.start || !selectionRange.end) return false;
    const dateStr = date.toISOString().split('T')[0];
    return dateStr >= selectionRange.start && dateStr <= selectionRange.end;
  };
  
  // Обробка виділення
  const handleMouseDown = (roomId: number, date: Date) => {
    setIsSelecting(true);
    setSelectionRange({
      start: date.toISOString().split('T')[0],
      end: date.toISOString().split('T')[0],
    });
    setSelectedRooms([roomId]);
  };
  
  const handleMouseEnter = (roomId: number, date: Date) => {
    if (!isSelecting) return;
    
    setSelectionRange(prev => ({
      ...prev,
      end: date.toISOString().split('T')[0],
    }));
    
    if (!selectedRooms.includes(roomId)) {
      setSelectedRooms(prev => [...prev, roomId]);
    }
  };
  
  const handleMouseUp = () => {
    if (isSelecting && selectionRange.start && selectionRange.end) {
      setShowCreateBooking(true);
    }
    setIsSelecting(false);
  };
  
  // Підказки
  const showTooltip = (event: React.MouseEvent, text: string) => {
    setTooltip({
      x: event.clientX + 15,
      y: event.clientY - 30,
      text,
      visible: true,
    });
  };
  
  const hideTooltip = () => {
    setTooltip({ ...tooltip, visible: false });
  };
  
  // Навігація
  const navigate = (direction: 'prev' | 'next') => {
    const newDate = new Date(currentDate);
    if (viewMode === 'month') {
      newDate.setMonth(newDate.getMonth() + (direction === 'next' ? 1 : -1));
    } else if (viewMode === 'week') {
      newDate.setDate(newDate.getDate() + (direction === 'next' ? 7 : -7));
    } else {
      newDate.setDate(newDate.getDate() + (direction === 'next' ? 1 : -1));
    }
    setCurrentDate(newDate);
  };
  
  // Стилі для статусів
  const cellStyles = {
    free: 'bg-green-50 hover:bg-green-100 border-green-200',
    occupied: 'bg-red-100 hover:bg-red-200 border-red-200 cursor-not-allowed',
    booked: 'bg-yellow-100 hover:bg-yellow-200 border-yellow-200',
    dirty: 'bg-orange-100 hover:bg-orange-200 border-orange-200',
    repair: 'bg-gray-200 hover:bg-gray-300 border-gray-300 cursor-not-allowed',
    blocked: 'bg-purple-100 hover:bg-purple-200 border-purple-200',
  };
  
  // Скидання фільтрів
  const resetFilters = () => {
    setSelectedCategories([]);
    setSearchQuery('');
    setShowOnlyAvailable(false);
    setSortBy('number');
  };
  
  // Підрахунок статистики за фільтрами
  const stats = {
    total: filteredRooms.length,
    available: filteredRooms.filter((r: Room) => r.status === 'clean').length,
    occupied: filteredRooms.filter((r: Room) => r.status === 'occupied').length,
    dirty: filteredRooms.filter((r: Room) => r.status === 'dirty').length,
  };
  
  if (roomsLoading || categoriesLoading) {
    return (
      <div className="flex items-center justify-center h-96">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-indigo-600"></div>
      </div>
    );
  }
  
  return (
    <div className="bg-white rounded-xl shadow-lg overflow-hidden">
      {/* Tooltip */}
      {tooltip.visible && (
        <div
          className="fixed z-50 bg-gray-900 text-white text-sm px-3 py-1.5 rounded-lg shadow-lg pointer-events-none whitespace-nowrap"
          style={{ left: tooltip.x, top: tooltip.y }}
        >
          {tooltip.text}
        </div>
      )}
      
      {/* Header */}
      <div className="border-b border-gray-200 bg-gradient-to-r from-gray-50 to-white px-6 py-4">
        <div className="flex flex-wrap items-center justify-between gap-4">
          <div className="flex items-center space-x-4">
            <button
              onClick={() => navigate('prev')}
              className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
            >
              <ChevronLeft className="w-5 h-5 text-gray-600" />
            </button>
            
            <h2 className="text-xl font-bold text-gray-800">
              {currentDate.toLocaleString('uk-UA', { month: 'long', year: 'numeric' })}
            </h2>
            
            <button
              onClick={() => navigate('next')}
              className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
            >
              <ChevronRight className="w-5 h-5 text-gray-600" />
            </button>
            
            <button
              onClick={() => setCurrentDate(new Date())}
              className="px-3 py-1.5 text-sm bg-indigo-50 text-indigo-600 rounded-lg hover:bg-indigo-100 transition-colors"
            >
              Сьогодні
            </button>
          </div>
          
          <div className="flex items-center space-x-3">
            <div className="flex bg-gray-100 rounded-lg p-1">
              {(['day', 'week', 'month'] as const).map((mode) => (
                <button
                  key={mode}
                  onClick={() => setViewMode(mode)}
                  className={`px-4 py-1.5 text-sm rounded-md transition-all ${
                    viewMode === mode
                      ? 'bg-white text-indigo-600 shadow-sm'
                      : 'text-gray-600 hover:bg-gray-200'
                  }`}
                >
                  {mode === 'day' ? 'День' : mode === 'week' ? 'Тиждень' : 'Місяць'}
                </button>
              ))}
            </div>
            
            {/* Кнопка фільтрів з індикатором активності */}
            <button
              onClick={() => setShowFilters(!showFilters)}
              className={`
                relative p-2 rounded-lg transition-colors
                ${showFilters || selectedCategories.length > 0 || searchQuery || showOnlyAvailable
                  ? 'bg-indigo-100 text-indigo-600'
                  : 'hover:bg-gray-100 text-gray-500'
                }
              `}
            >
              <Filter className="w-5 h-5" />
              {(selectedCategories.length > 0 || searchQuery || showOnlyAvailable) && (
                <span className="absolute -top-1 -right-1 w-3 h-3 bg-indigo-600 rounded-full"></span>
              )}
            </button>
            
            <button className="p-2 hover:bg-gray-100 rounded-lg transition-colors">
              <Download className="w-5 h-5 text-gray-500" />
            </button>
          </div>
        </div>
        
        {/* Панель фільтрів */}
        {showFilters && (
          <div className="mt-4 p-4 bg-gray-50 rounded-xl border border-gray-200 animate-in slide-in-from-top-2 duration-200">
            <div className="flex items-center justify-between mb-4">
              <h3 className="font-medium text-gray-800">Фільтри номерів</h3>
              <button
                onClick={resetFilters}
                className="text-sm text-indigo-600 hover:text-indigo-800 flex items-center space-x-1"
              >
                <X className="w-3 h-3" />
                <span>Скинути всі</span>
              </button>
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
              {/* Пошук за номером */}
              <div>
                <label className="block text-xs font-medium text-gray-500 mb-1">
                  Пошук за номером
                </label>
                <div className="relative">
                  <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 w-4 h-4 text-gray-400" />
                  <input
                    type="text"
                    value={searchQuery}
                    onChange={(e) => setSearchQuery(e.target.value)}
                    placeholder="101, 202A..."
                    className="w-full pl-9 pr-3 py-2 text-sm border border-gray-200 rounded-lg focus:ring-2 focus:ring-indigo-500 focus:border-transparent"
                  />
                </div>
              </div>
              
              {/* Сортування */}
              <div>
                <label className="block text-xs font-medium text-gray-500 mb-1">
                  Сортувати за
                </label>
                <select
                  value={sortBy}
                  onChange={(e) => setSortBy(e.target.value as any)}
                  className="w-full px-3 py-2 text-sm border border-gray-200 rounded-lg focus:ring-2 focus:ring-indigo-500"
                >
                  <option value="number">Номером (зростання)</option>
                  <option value="floor">Поверхом</option>
                  <option value="capacity">Місткістю</option>
                  <option value="price">Ціною</option>
                </select>
              </div>
              
              {/* Тільки вільні */}
              <div className="flex items-end">
                <label className="flex items-center space-x-2 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={showOnlyAvailable}
                    onChange={(e) => setShowOnlyAvailable(e.target.checked)}
                    className="w-4 h-4 text-indigo-600 rounded border-gray-300 focus:ring-indigo-500"
                  />
                  <span className="text-sm text-gray-700">Показати тільки вільні номери</span>
                </label>
              </div>
              
              {/* Статистика */}
              <div className="flex items-center justify-end space-x-4 text-sm">
                <div className="flex items-center space-x-1">
                  <div className="w-2 h-2 bg-green-500 rounded-full"></div>
                  <span className="text-gray-600">Вільні: {stats.available}</span>
                </div>
                <div className="flex items-center space-x-1">
                  <div className="w-2 h-2 bg-red-500 rounded-full"></div>
                  <span className="text-gray-600">Зайняті: {stats.occupied}</span>
                </div>
                <div className="flex items-center space-x-1">
                  <div className="w-2 h-2 bg-orange-500 rounded-full"></div>
                  <span className="text-gray-600">Брудні: {stats.dirty}</span>
                </div>
              </div>
            </div>
            
            {/* Категорії */}
            <div className="mt-4">
              <label className="block text-xs font-medium text-gray-500 mb-2">
                Категорії номерів
              </label>
              <div className="flex flex-wrap gap-2">
                {categories?.data?.map((category: Category) => {
                  const Icon = categoryIcons[category.code] || categoryIcons.default;
                  const isSelected = selectedCategories.includes(category.id);
                  const categoryCount = rooms?.data?.filter((r: Room) => r.categoryId === category.id).length || 0;
                  
                  return (
                    <button
                      key={category.id}
                      onClick={() => {
                        setSelectedCategories(prev =>
                          isSelected
                            ? prev.filter(id => id !== category.id)
                            : [...prev, category.id]
                        );
                      }}
                      className={`
                        flex items-center space-x-2 px-3 py-1.5 rounded-lg text-sm transition-all
                        ${isSelected
                          ? categoryColors[category.code] || categoryColors.default
                          : 'bg-gray-100 text-gray-600 hover:bg-gray-200'
                        }
                      `}
                    >
                      <Icon className="w-3.5 h-3.5" />
                      <span>{category.name}</span>
                      <span className="text-xs opacity-70">({categoryCount})</span>
                    </button>
                  );
                })}
              </div>
            </div>
          </div>
        )}
        
        {/* Легенда */}
        <div className="flex flex-wrap gap-4 mt-4 text-xs">
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-green-100 border border-green-200 rounded"></div>
            <span className="text-gray-600">Вільний</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-yellow-100 border border-yellow-200 rounded"></div>
            <span className="text-gray-600">Заброньовано</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-red-100 border border-red-200 rounded"></div>
            <span className="text-gray-600">Заселено</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-orange-100 border border-orange-200 rounded"></div>
            <span className="text-gray-600">Потребує прибирання</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-gray-200 border border-gray-300 rounded"></div>
            <span className="text-gray-600">На ремонті</span>
          </div>
          <div className="flex items-center space-x-1.5 ml-4">
            <div className="w-3 h-3 bg-indigo-200 border border-indigo-300 rounded"></div>
            <span className="text-gray-600">Виділено для бронювання</span>
          </div>
        </div>
      </div>
      
      {/* Шахівка */}
      <div
        ref={chessboardRef}
        className="overflow-x-auto"
        onMouseUp={handleMouseUp}
      >
        <div className="min-w-[800px]">
          {/* Заголовок з датами */}
          <div className="flex border-b border-gray-200 bg-gray-50">
            <div className="w-40 flex-shrink-0 p-3 text-left text-sm font-medium text-gray-500 border-r border-gray-200">
              Номер / Категорія
            </div>
            <div className="flex-1 grid" style={{ gridTemplateColumns: `repeat(${days.length}, minmax(100px, 1fr))` }}>
              {days.map((date, idx) => (
                <div
                  key={idx}
                  className={`p-2 text-center border-r border-gray-200 ${
                    date.toDateString() === new Date().toDateString() ? 'bg-indigo-50' : ''
                  }`}
                >
                  <div className="text-sm font-medium text-gray-700">
                    {date.toLocaleString('uk-UA', { day: 'numeric', month: 'short' })}
                  </div>
                  <div className="text-xs text-gray-400">
                    {date.toLocaleString('uk-UA', { weekday: 'short' })}
                  </div>
                </div>
              ))}
            </div>
          </div>
          
          {/* Рядки з номерами */}
          {filteredRooms.map((room: Room) => {
            const CategoryIcon = categoryIcons[room.category?.code] || categoryIcons.default;
            const categoryColor = categoryColors[room.category?.code] || categoryColors.default;
            
            return (
              <div key={room.id} className="flex border-b border-gray-100 hover:bg-gray-50 transition-colors group">
                {/* Інформація про номер */}
                <div className="w-40 flex-shrink-0 p-3 border-r border-gray-200">
                  <div className="flex items-center justify-between">
                    <div>
                      <div className="font-medium text-gray-800">{room.number}</div>
                      <div className="text-xs text-gray-400 flex items-center space-x-1 mt-0.5">
                        <Bed className="w-3 h-3" />
                        <span>{room.capacity}</span>
                      </div>
                    </div>
                    <div className={`
                      flex items-center space-x-1 px-2 py-0.5 rounded-full text-xs
                      ${categoryColor}
                    `}>
                      <CategoryIcon className="w-3 h-3" />
                      <span>{room.category?.name}</span>
                    </div>
                  </div>
                  <div className="text-xs text-gray-400 mt-1 truncate">
                    {room.name}
                  </div>
                </div>
                
                {/* Клітинки днів */}
                <div
                  className="flex-1 grid"
                  style={{ gridTemplateColumns: `repeat(${days.length}, minmax(100px, 1fr))` }}
                >
                  {days.map((date, idx) => {
                    const booking = getBookingForCell(room.id, date);
                    const status = getCellStatus(room, date, booking);
                    const isSelected = isInSelectionRange(room.id, date);
                    const isHovered = hoveredCell?.roomId === room.id && hoveredCell?.date === date.toISOString().split('T')[0];
                    
                    const statusStyle = cellStyles[status as keyof typeof cellStyles];
                    const isInteractive = status === 'free' || status === 'dirty';
                    
                    return (
                      <div
                        key={idx}
                        ref={(el) => {
                          if (el) {
                            cellRefs.current.set(`${room.id}-${date.toISOString().split('T')[0]}`, el);
                          }
                        }}
                        className={`
                          relative p-2 border-r border-b border-gray-100 transition-all duration-150
                          ${statusStyle}
                          ${isSelected ? 'ring-2 ring-indigo-400 ring-inset bg-indigo-100' : ''}
                          ${isHovered ? 'scale-[0.98] shadow-inner' : ''}
                          ${isInteractive ? 'cursor-pointer' : 'cursor-not-allowed opacity-70'}
                        `}
                        onMouseDown={() => isInteractive && handleMouseDown(room.id, date)}
                        onMouseEnter={(e) => {
                          if (isInteractive) {
                            handleMouseEnter(room.id, date);
                            setHoveredCell({ roomId: room.id, date: date.toISOString().split('T')[0] });
                          }
                          if (booking) {
                            showTooltip(e, `${booking.guestName}\n📅 ${booking.checkIn} → ${booking.checkOut}\n👥 ${booking.adults} дор., ${booking.children} діт.\n💰 ${booking.totalAmount} ₴`);
                          } else {
                            showTooltip(e, `${room.number} (${room.category?.name})\n✨ Вільний\n💰 ${room.category?.basePrice} ₴/ніч\n🖱️ Натисніть та протягніть для бронювання`);
                          }
                        }}
                        onMouseLeave={() => {
                          hideTooltip();
                          setHoveredCell(null);
                        }}
                      >
                        {booking ? (
                          <div className="text-xs">
                            <div className="font-medium text-gray-800 truncate">
                              {booking.guestName}
                            </div>
                            <div className="text-gray-500 text-[10px] mt-0.5">
                              {booking.adults} дор. • {booking.children} діт.
                            </div>
                            {booking.status === 'checked_in' && (
                              <div className="mt-1 text-[10px] text-red-500 font-medium">
                                Заселено
                              </div>
                            )}
                          </div>
                        ) : (
                          <div className="text-center text-gray-400 text-[10px]">
                            {status === 'free' && '✓ Вільний'}
                            {status === 'dirty' && '🧹 Потребує прибирання'}
                            {status === 'repair' && '🔧 На ремонті'}
                          </div>
                        )}
                        
                        {/* Індикатор діапазону */}
                        {isSelected && selectionRange.start && selectionRange.end && (
                          <div className="absolute inset-0 bg-indigo-100 opacity-30 pointer-events-none" />
                        )}
                      </div>
                    );
                  })}
                </div>
              </div>
            );
          })}
          
          {/* Повідомлення про відсутність результатів */}
          {filteredRooms.length === 0 && (
            <div className="flex flex-col items-center justify-center py-16 text-gray-400">
              <Filter className="w-12 h-12 mb-3 opacity-50" />
              <p className="text-lg font-medium">Немає номерів за вибраними фільтрами</p>
              <button
                onClick={resetFilters}
                className="mt-3 text-indigo-600 hover:text-indigo-800 text-sm"
              >
                Скинути фільтри
              </button>
            </div>
          )}
        </div>
      </div>
      
      {/* Модальне вікно створення бронювання */}
      {showCreateBooking && selectionRange.start && selectionRange.end && (
        <CreateBookingModal
          propertyId={propertyId}
          checkIn={selectionRange.start}
          checkOut={selectionRange.end}
          roomIds={selectedRooms}
          onClose={() => {
            setShowCreateBooking(false);
            setSelectionRange({ start: null, end: null });
            setSelectedRooms([]);
          }}
          onSubmit={(data) => {
            // Мутація створення бронювання
            console.log('Submit booking:', data);
            setShowCreateBooking(false);
            setSelectionRange({ start: null, end: null });
            setSelectedRooms([]);
            toast.success('Бронювання створено!');
            refetchBookings();
          }}
          isLoading={false}
        />
      )}
    </div>
  );
}

// server/routes/properties.ts (оновлений)

import { Router } from 'express';
import { db } from '../db';
import { properties, rooms, categories, bookings } from '../db/schema';
import { eq, and, like, sql } from 'drizzle-orm';

const router = Router();

// GET /api/properties/:id/categories
router.get('/properties/:id/categories', async (req, res) => {
  const propertyId = parseInt(req.params.id);
  
  const categoriesList = await db.select()
    .from(categories)
    .where(eq(categories.propertyId, propertyId))
    .orderBy(categories.name);
  
  res.json({ success: true, data: categoriesList });
});

// GET /api/properties/:id/rooms (з фільтрами)
router.get('/properties/:id/rooms', async (req, res) => {
  const propertyId = parseInt(req.params.id);
  const { categories: categoryIds, search, sortBy, onlyAvailable } = req.query;
  
  let query = db.select()
    .from(rooms)
    .where(eq(rooms.propertyId, propertyId));
  
  // Фільтр за категоріями
  if (categoryIds && typeof categoryIds === 'string' && categoryIds.length > 0) {
    const ids = categoryIds.split(',').map(Number);
    query = query.where(sql`${rooms.categoryId} IN (${ids.join(',')})`);
  }
  
  // Пошук за номером
  if (search && typeof search === 'string' && search.length > 0) {
    query = query.where(like(rooms.number, `%${search}%`));
  }
  
  // Тільки вільні номери
  if (onlyAvailable === 'true') {
    query = query.where(eq(rooms.status, 'clean'));
  }
  
  // Сортування
  switch (sortBy) {
    case 'floor':
      query = query.orderBy(rooms.floor, rooms.number);
      break;
    case 'capacity':
      query = query.orderBy(rooms.capacity, rooms.number);
      break;
    case 'price':
      query = query.orderBy(rooms.basePrice, rooms.number);
      break;
    default:
      query = query.orderBy(rooms.number);
  }
  
  const roomsList = await query;
  
  // Додаємо інформацію про категорію
  const roomsWithCategories = await Promise.all(
    roomsList.map(async (room) => {
      const category = await db.select()
        .from(categories)
        .where(eq(categories.id, room.categoryId))
        .get();
      
      return {
        ...room,
        category,
      };
    })
  );
  
  res.json({ success: true, data: roomsWithCategories });
});

// POST /api/properties/:id/categories (створення категорії)
router.post('/properties/:id/categories', async (req, res) => {
  const propertyId = parseInt(req.params.id);
  const { name, code, basePrice, capacity, color, icon } = req.body;
  
  const [category] = await db.insert(categories).values({
    propertyId,
    name,
    code,
    basePrice,
    capacity: capacity || 2,
    color: color || '#4f46e5',
    icon: icon || 'bed',
  }).returning();
  
  res.json({ success: true, data: category });
});

export default router;
