import { Request, Response } from 'express';
import { prisma } from '../index';

// Отримати всі об'єкти (тільки для SUPER_ADMIN)
export const getAllProperties = async (req: Request, res: Response) => {
  try {
    const properties = await prisma.property.findMany({
      include: {
        _count: {
          select: {
            users: true,
            roomTypes: true,
            bookings: true,
          },
        },
      },
    });
    res.json(properties);
  } catch (error) {
    console.error('Get properties error:', error);
    res.status(500).json({ error: 'Помилка отримання списку об\'єктів' });
  }
};

// Отримати один об'єкт
export const getPropertyById = async (req: Request, res: Response) => {
  try {
    const { id } = req.params;
    const property = await prisma.property.findUnique({
      where: { id },
      include: {
        roomTypes: {
          include: { rooms: true },
        },
        bookings: {
          take: 10,
          orderBy: { createdAt: 'desc' },
        },
      },
    });

    if (!property) {
      res.status(404).json({ error: 'Об\'єкт не знайдено' });
      return;
    }

    res.json(property);
  } catch (error) {
    console.error('Get property error:', error);
    res.status(500).json({ error: 'Помилка отримання об\'єкта' });
  }
};

// Створити новий об'єкт
export const createProperty = async (req: Request, res: Response) => {
  try {
    const { name, address, phone, email, logo, checkInTime, checkOutTime, description } = req.body;

    const property = await prisma.property.create({
      data: {
        name,
        address,
        phone,
        email,
        logo,
        checkInTime: checkInTime || '14:00',
        checkOutTime: checkOutTime || '12:00',
        description,
        settings: {},
      },
    });

    res.status(201).json(property);
  } catch (error) {
    console.error('Create property error:', error);
    res.status(500).json({ error: 'Помилка створення об\'єкта' });
  }
};

// Оновити об'єкт
export const updateProperty = async (req: Request, res: Response) => {
  try {
    const { id } = req.params;
    const { name, address, phone, email, logo, checkInTime, checkOutTime, description, settings } = req.body;

    const property = await prisma.property.update({
      where: { id },
      data: {
        name,
        address,
        phone,
        email,
        logo,
        checkInTime,
        checkOutTime,
        description,
        settings,
      },
    });

    res.json(property);
  } catch (error) {
    console.error('Update property error:', error);
    res.status(500).json({ error: 'Помилка оновлення об\'єкта' });
  }
};

// Видалити об'єкт
export const deleteProperty = async (req: Request, res: Response) => {
  try {
    const { id } = req.params;
    await prisma.property.delete({ where: { id } });
    res.json({ message: 'Об\'єкт успішно видалено' });
  } catch (error) {
    console.error('Delete property error:', error);
    res.status(500).json({ error: 'Помилка видалення об\'єкта' });
  }
};
