"""
Housekeeping model - служба покоївок
"""

from datetime import datetime
from app import db

class HousekeepingTask(db.Model):
    """Модель завдання для покоївки"""
    __tablename__ = 'housekeeping_tasks'
    
    id = db.Column(db.Integer, primary_key=True)
    
    # Номер
    room_id = db.Column(db.Integer, db.ForeignKey('rooms.id'), nullable=False)
    room = db.relationship('Room', backref='housekeeping_tasks')
    
    # Тип завдання
    task_type = db.Column(db.String(20), nullable=False)
    # cleaning, deep_cleaning, linen_change, maintenance, check
    
    # Пріоритет
    priority = db.Column(db.String(20), default='normal')
    # high, normal, low
    
    # Кому призначено
    assigned_to = db.Column(db.Integer, db.ForeignKey('users.id'))
    assignee = db.relationship('User', foreign_keys=[assigned_to], backref='assigned_tasks')
    assigned_by = db.Column(db.Integer, db.ForeignKey('users.id'))
    assigner = db.relationship('User', foreign_keys=[assigned_by])
    
    # Статус
    status = db.Column(db.String(20), default='pending')
    # pending, in_progress, completed, verified, cancelled
    
    # Час
    start_time = db.Column(db.DateTime)
    end_time = db.Column(db.DateTime)
    estimated_minutes = db.Column(db.Integer, default=30)
    
    # Перевірка контролером
    verified_by = db.Column(db.Integer, db.ForeignKey('users.id'))
    verified_at = db.Column(db.DateTime)
    verification_notes = db.Column(db.Text)
    
    # Додатково
    notes = db.Column(db.Text)
    supplies_used = db.Column(db.JSON, default=list)  # використані засоби
    
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, onupdate=datetime.utcnow)
    
    @property
    def status_display(self):
        """Відображення статусу українською"""
        status_map = {
            'pending': 'Очікує',
            'in_progress': 'В процесі',
            'completed': 'Виконано',
            'verified': 'Перевірено',
            'cancelled': 'Скасовано'
        }
        return status_map.get(self.status, self.status)
    
    @property
    def type_display(self):
        """Відображення типу українською"""
        type_map = {
            'cleaning': 'Прибирання',
            'deep_cleaning': 'Генеральне прибирання',
            'linen_change': 'Зміна білизни',
            'maintenance': 'Обслуговування',
            'check': 'Перевірка'
        }
        return type_map.get(self.task_type, self.task_type)
    
    def start(self):
        """Початок виконання"""
        if self.status != 'pending':
            raise ValueError(f"Cannot start task with status {self.status}")
        
        self.status = 'in_progress'
        self.start_time = datetime.utcnow()
        db.session.commit()
    
    def complete(self):
        """Завершення виконання"""
        if self.status != 'in_progress':
            raise ValueError(f"Cannot complete task with status {self.status}")
        
        self.status = 'completed'
        self.end_time = datetime.utcnow()
        db.session.commit()
    
    def verify(self, user_id, notes=None):
        """Перевірка контролером"""
        if self.status != 'completed':
            raise ValueError(f"Cannot verify task with status {self.status}")
        
        self.status = 'verified'
        self.verified_by = user_id
        self.verified_at = datetime.utcnow()
        self.verification_notes = notes
        db.session.commit()
        
        # Оновлення статусу номера
        if self.room:
            self.room.status = 'clean'
            db.session.commit()
    
    def to_dict(self):
        return {
            'id': self.id,
            'room_id': self.room_id,
            'room_number': self.room.number if self.room else None,
            'task_type': self.task_type,
            'type_display': self.type_display,
            'priority': self.priority,
            'status': self.status,
            'status_display': self.status_display,
            'assigned_to': self.assigned_to,
            'assignee_name': self.assignee.full_name if self.assignee else None,
            'start_time': self.start_time.isoformat() if self.start_time else None,
            'end_time': self.end_time.isoformat() if self.end_time else None,
            'estimated_minutes': self.estimated_minutes,
            'notes': self.notes,
            'created_at': self.created_at.isoformat()
        }

"""
Night Audit Service - автоматичне підбиття підсумків дня
"""

from datetime import datetime, timedelta, date
from app import db
from models.audit import NightAudit, AuditLog
from models.reservation import Reservation
from models.room import Room
from models.invoice import Invoice
from loguru import logger


class NightAuditService:
    """Сервіс для виконання нічного аудиту"""
    
    def __init__(self, hotel_id, user_id=None):
        self.hotel_id = hotel_id
        self.user_id = user_id
        self.audit_date = date.today() - timedelta(days=1)
        self.errors = []
        self.warnings = []
    
    def run(self):
        """Виконання нічного аудиту"""
        logger.info(f"Starting night audit for {self.audit_date}")
        
        # Перевірка чи аудит вже виконано
        existing = NightAudit.query.filter_by(audit_date=self.audit_date).first()
        if existing and existing.status == 'completed':
            logger.warning(f"Night audit for {self.audit_date} already completed")
            return existing
        
        # Створення запису аудиту
        audit = NightAudit(
            audit_date=self.audit_date,
            status='pending'
        )
        db.session.add(audit)
        db.session.commit()
        
        try:
            # 1. Збір статистики по номерах
            self._collect_room_stats(audit)
            
            # 2. Збір фінансової статистики
            self._collect_financial_stats(audit)
            
            # 3. Обробка заїздів/виїздів
            self._process_movements(audit)
            
            # 4. Перевірка неузгодженостей
            self._check_discrepancies(audit)
            
            # 5. Закриття дня
            self._close_day(audit)
            
            audit.status = 'completed'
            audit.performed_by = self.user_id
            audit.performed_at = datetime.utcnow()
            audit.errors = self.errors
            audit.warnings = self.warnings
            
            db.session.commit()
            
            logger.info(f"Night audit completed for {self.audit_date}")
            
            # Відправка звіту в Telegram
            self._send_report(audit)
            
            return audit
            
        except Exception as e:
            logger.error(f"Night audit failed: {e}")
            audit.status = 'failed'
            audit.errors.append(str(e))
            db.session.commit()
            raise
    
    def _collect_room_stats(self, audit):
        """Збір статистики по номерах"""
        total_rooms = Room.query.filter_by(hotel_id=self.hotel_id).count()
        occupied_rooms = Room.query.filter_by(
            hotel_id=self.hotel_id,
            status='occupied'
        ).count()
        
        audit.total_rooms = total_rooms
        audit.occupied_rooms = occupied_rooms
        audit.occupancy_rate = (occupied_rooms / total_rooms * 100) if total_rooms > 0 else 0
        
        # Перевірка на аномалії
        if audit.occupancy_rate > 100:
            self.warnings.append(f"Occupancy rate > 100%: {audit.occupancy_rate}")
    
    def _collect_financial_stats(self, audit):
        """Збір фінансової статистики за день"""
        start_datetime = datetime.combine(self.audit_date, datetime.min.time())
        end_datetime = datetime.combine(self.audit_date, datetime.max.time())
        
        # Рахунки за день
        invoices = Invoice.query.filter(
            Invoice.created_at >= start_datetime,
            Invoice.created_at <= end_datetime,
            Invoice.status == 'paid'
        ).all()
        
        total = sum(i.amount for i in invoices)
        accommodation = sum(i.amount for i in invoices if i.invoice_type == 'accommodation')
        services = sum(i.amount for i in invoices if i.invoice_type == 'service')
        
        audit.total_revenue = total
        audit.accommodation_revenue = accommodation
        audit.services_revenue = services
        
        # Перевірка на великі суми
        if total > 100000:
            self.warnings.append(f"High revenue today: {total} UAH")
    
    def _process_movements(self, audit):
        """Обробка заїздів та виїздів за день"""
        start_datetime = datetime.combine(self.audit_date, datetime.min.time())
        end_datetime = datetime.combine(self.audit_date, datetime.max.time())
        
        # Заїзди
        checkins = Reservation.query.filter(
            Reservation.hotel_id == self.hotel_id,
            Reservation.actual_check_in >= start_datetime,
            Reservation.actual_check_in <= end_datetime
        ).count()
        
        # Виїзди
        checkouts = Reservation.query.filter(
            Reservation.hotel_id == self.hotel_id,
            Reservation.actual_check_out >= start_datetime,
            Reservation.actual_check_out <= end_datetime
        ).count()
        
        # Нові бронювання
        new_reservations = Reservation.query.filter(
            Reservation.hotel_id == self.hotel_id,
            Reservation.created_at >= start_datetime,
            Reservation.created_at <= end_datetime
        ).count()
        
        # Скасування
        cancellations = Reservation.query.filter(
            Reservation.hotel_id == self.hotel_id,
            Reservation.status == 'cancelled',
            Reservation.updated_at >= start_datetime,
            Reservation.updated_at <= end_datetime
        ).count()
        
        # Загальна кількість гостей
        total_guests = Reservation.query.filter(
            Reservation.hotel_id == self.hotel_id,
            Reservation.status == 'checked_in'
        ).count()
        
        audit.new_checkins = checkins
        audit.checkouts = checkouts
        audit.new_reservations = new_reservations
        audit.cancellations = cancellations
        audit.total_guests = total_guests
    
    def _check_discrepancies(self, audit):
        """Перевірка неузгодженостей"""
        # Перевірка чи всі виселення мають закриті рахунки
        checkouts = Reservation.query.filter(
            Reservation.hotel_id == self.hotel_id,
            Reservation.status == 'checked_out',
            Reservation.actual_check_out >= datetime.combine(self.audit_date, datetime.min.time()),
            Reservation.actual_check_out <= datetime.combine(self.audit_date, datetime.max.time())
        ).all()
        
        for reservation in checkouts:
            if reservation.balance > 0:
                self.errors.append(
                    f"Reservation {reservation.reservation_number} has unpaid balance: {reservation.balance} UAH"
                )
    
    def _close_day(self, audit):
        """Закриття дня - автоматичні дії"""
        # Оновлення статусів номерів
        rooms_to_clean = Room.query.filter_by(
            hotel_id=self.hotel_id,
            status='occupied'
        ).all()
        
        for room in rooms_to_clean:
            room.status = 'dirty'
        
        # Створення завдань для покоївок на наступний день
        self._create_housekeeping_tasks()
        
        db.session.commit()
    
    def _create_housekeeping_tasks(self):
        """Створення завдань для покоївок на наступний день"""
        from models.housekeeping import HousekeepingTask
        
        # Номери, які потребують прибирання
        dirty_rooms = Room.query.filter_by(
            hotel_id=self.hotel_id,
            status='dirty'
        ).all()
        
        for room in dirty_rooms:
            existing = HousekeepingTask.query.filter(
                HousekeepingTask.room_id == room.id,
                HousekeepingTask.status.in_(['pending', 'in_progress']),
                HousekeepingTask.created_at >= datetime.now() - timedelta(days=1)
            ).first()
            
            if not existing:
                task = HousekeepingTask(
                    room_id=room.id,
                    task_type='cleaning',
                    priority='normal'
                )
                db.session.add(task)
    
    def _send_report(self, audit):
        """Відправка звіту в Telegram"""
        from services.telegram_bot import send_notification
        
        report = f"""
📊 <b>Нічний аудит {audit.audit_date}</b>

🏨 <b>Заповненість:</b> {audit.occupancy_rate:.1f}% ({audit.occupied_rooms}/{audit.total_rooms})

💰 <b>Дохід за день:</b> {audit.total_revenue:,.2f} ₴
   • Проживання: {audit.accommodation_revenue:,.2f} ₴
   • Послуги: {audit.services_revenue:,.2f} ₴

👥 <b>Рух гостей:</b>
   • Заїзди: +{audit.new_checkins}
   • Виїзди: -{audit.checkouts}
   • Всього гостей: {audit.total_guests}

📅 <b>Бронювання:</b>
   • Нові: +{audit.new_reservations}
   • Скасування: -{audit.cancellations}
"""
        
        if self.warnings:
            report += f"\n⚠️ <b>Попередження:</b>\n" + "\n".join(self.warnings[:3])
        
        if self.errors:
            report += f"\n❌ <b>Помилки:</b>\n" + "\n".join(self.errors[:3])
        
        send_notification(report)

"""
1C Export Service - експорт даних для 1С:Бухгалтерія
"""

import json
import csv
from datetime import datetime, timedelta
from io import StringIO
from models.reservation import Reservation
from models.invoice import Invoice
from models.guest import Guest, Company
from loguru import logger


class Export1CService:
    """Сервіс для експорту даних в 1С"""
    
    def __init__(self, hotel_id, date_from, date_to):
        self.hotel_id = hotel_id
        self.date_from = datetime.strptime(date_from, '%Y-%m-%d') if isinstance(date_from, str) else date_from
        self.date_to = datetime.strptime(date_to, '%Y-%m-%d') if isinstance(date_to, str) else date_to
    
    def export_invoices(self, format='json'):
        """Експорт рахунків за період"""
        invoices = Invoice.query.filter(
            Invoice.created_at >= self.date_from,
            Invoice.created_at <= self.date_to,
            Invoice.status == 'paid'
        ).all()
        
        data = []
        for inv in invoices:
            data.append({
                'НомерДокумента': inv.invoice_number,
                'ДатаДокумента': inv.created_at.strftime('%Y-%m-%d'),
                'Сумма': inv.amount,
                'НДС': inv.tax,
                'Контрагент': self._get_contractor_name(inv),
                'ВидОперации': inv.invoice_type
            })
        
        if format == 'json':
            return json.dumps(data, ensure_ascii=False, indent=2)
        elif format == 'csv':
            return self._to_csv(data)
        else:
            return data
    
    def export_guests(self):
        """Експорт гостей для реєстрації"""
        guests = Guest.query.filter(
            Guest.created_at >= self.date_from,
            Guest.created_at <= self.date_to
        ).all()
        
        data = []
        for guest in guests:
            data.append({
                'Прізвище': guest.last_name,
                'Ім\'я': guest.first_name,
                'По_батькові': guest.middle_name or '',
                'Паспорт': guest.passport_number,
                'Дата_народження': guest.birth_date.strftime('%Y-%m-%d') if guest.birth_date else '',
                'Адреса': guest.address or '',
                'Телефон': guest.phone
            })
        
        return json.dumps(data, ensure_ascii=False, indent=2)
    
    def export_daily_report(self):
        """Експорт денного звіту для 1С"""
        invoices = Invoice.query.filter(
            Invoice.created_at >= self.date_from,
            Invoice.created_at <= self.date_to
        ).all()
        
        # Групування за типом оплати
        by_payment_method = {}
        for inv in invoices:
            method = inv.payment_method or 'unknown'
            by_payment_method[method] = by_payment_method.get(method, 0) + inv.amount
        
        report = {
            'Період': f"{self.date_from.strftime('%Y-%m-%d')} - {self.date_to.strftime('%Y-%m-%d')}",
            'Загальний_дохід': sum(i.amount for i in invoices),
            'Кількість_операцій': len(invoices),
            'Деталізація_за_типом_оплати': by_payment_method
        }
        
        return json.dumps(report, ensure_ascii=False, indent=2)
    
    def _get_contractor_name(self, invoice):
        """Отримання назви контрагента для рахунку"""
        if invoice.company_id:
            company = Company.query.get(invoice.company_id)
            if company:
                return company.name
        
        if invoice.guest_id:
            guest = Guest.query.get(invoice.guest_id)
            if guest:
                return guest.full_name
        
        return 'Фізична особа'
    
    def _to_csv(self, data):
        """Конвертація в CSV формат"""
        if not data:
            return ''
        
        output = StringIO()
        writer = csv.DictWriter(output, fieldnames=data[0].keys())
        writer.writeheader()
        writer.writerows(data)
        
        return output.getvalue()

@api_bp.route('/invoices/<int:invoice_id>/pay', methods=['POST'])
@jwt_required()
def pay_invoice(invoice_id):
    """Оплата рахунку"""
    invoice = Invoice.query.get_or_404(invoice_id)
    data = request.get_json()
    
    if invoice.status == 'paid':
        return jsonify({'error': 'Invoice already paid'}), 400
    
    amount = data.get('amount', invoice.amount)
    payment_method = data.get('payment_method', 'cash')
    payment_details = data.get('payment_details', {})
    
    invoice.paid += amount
    invoice.payment_method = payment_method
    invoice.payment_details = payment_details
    
    if invoice.paid >= invoice.amount:
        invoice.status = 'paid'
        invoice.paid_at = datetime.utcnow()
    
    db.session.commit()
    
    return jsonify(invoice.to_dict()), 200

    total = sum(i.amount for i in invoices)
    paid = sum(i.paid for i in invoices)
    
    return jsonify({
        'invoices': [i.to_dict() for i in invoices],
        'total_amount': total,
        'paid_amount': paid,
        'balance': total - paid
    }), 200

// Service Worker для PWA
const CACHE_NAME = 'maison-hd-v1';
const urlsToCache = [
  '/',
  '/static/css/style.css',
  '/static/js/main.js'
];

self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => cache.addAll(urlsToCache))
  );
});

self.addEventListener('fetch', event => {
  event.respondWith(
    caches.match(event.request)
      .then(response => response || fetch(event.request))
  );
});

self.addEventListener('push', event => {
  const data = event.data.json();
  const options = {
    body: data.body,
    icon: '/static/assets/icon-192.png',
    badge: '/static/assets/badge.png',
    vibrate: [200, 100, 200],
    data: {
      url: data.url || '/'
    }
  };
  
  event.waitUntil(
    self.registration.showNotification(data.title, options)
  );
});

self.addEventListener('notificationclick', event => {
  event.notification.close();
  event.waitUntil(
    clients.openWindow(event.notification.data.url)
  );
});
