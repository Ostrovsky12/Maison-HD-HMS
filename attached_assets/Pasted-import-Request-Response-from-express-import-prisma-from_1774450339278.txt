import { Request, Response } from 'express';
import { prisma } from '../index';
import { RoomStatus } from '@prisma/client';

// ============ Room Types ============

export const getRoomTypes = async (req: Request, res: Response) => {
  try {
    const { propertyId } = req.query;
    
    const where = propertyId ? { propertyId: propertyId as string } : {};
    
    const roomTypes = await prisma.roomType.findMany({
      where,
      include: {
        rooms: true,
        property: true,
      },
    });
    
    res.json(roomTypes);
  } catch (error) {
    console.error('Get room types error:', error);
    res.status(500).json({ error: 'Помилка отримання типів номерів' });
  }
};

export const createRoomType = async (req: Request, res: Response) => {
  try {
    const { propertyId, name, description, capacity, amenities, images, basePrice } = req.body;
    
    const roomType = await prisma.roomType.create({
      data: {
        propertyId,
        name,
        description,
        capacity,
        amenities: amenities || [],
        images: images || [],
        basePrice,
      },
    });
    
    res.status(201).json(roomType);
  } catch (error) {
    console.error('Create room type error:', error);
    res.status(500).json({ error: 'Помилка створення типу номера' });
  }
};

export const updateRoomType = async (req: Request, res: Response) => {
  try {
    const { id } = req.params;
    const { name, description, capacity, amenities, images, basePrice } = req.body;
    
    const roomType = await prisma.roomType.update({
      where: { id },
      data: { name, description, capacity, amenities, images, basePrice },
    });
    
    res.json(roomType);
  } catch (error) {
    console.error('Update room type error:', error);
    res.status(500).json({ error: 'Помилка оновлення типу номера' });
  }
};

export const deleteRoomType = async (req: Request, res: Response) => {
  try {
    const { id } = req.params;
    await prisma.roomType.delete({ where: { id } });
    res.json({ message: 'Тип номера успішно видалено' });
  } catch (error) {
    console.error('Delete room type error:', error);
    res.status(500).json({ error: 'Помилка видалення типу номера' });
  }
};

// ============ Rooms ============

export const getRooms = async (req: Request, res: Response) => {
  try {
    const { propertyId, roomTypeId, status } = req.query;
    
    const where: any = {};
    if (propertyId) where.roomType = { propertyId: propertyId as string };
    if (roomTypeId) where.roomTypeId = roomTypeId as string;
    if (status) where.status = status as RoomStatus;
    
    const rooms = await prisma.room.findMany({
      where,
      include: {
        roomType: true,
        bookings: {
          where: {
            status: { in: ['CONFIRMED', 'CHECKED_IN'] },
            checkOut: { gte: new Date() },
          },
          take: 1,
        },
      },
    });
    
    res.json(rooms);
  } catch (error) {
    console.error('Get rooms error:', error);
    res.status(500).json({ error: 'Помилка отримання номерів' });
  }
};

export const createRoom = async (req: Request, res: Response) => {
  try {
    const { roomTypeId, number, floor, description } = req.body;
    
    const room = await prisma.room.create({
      data: {
        roomTypeId,
        number,
        floor,
        description,
        status: 'AVAILABLE',
      },
    });
    
    res.status(201).json(room);
  } catch (error) {
    console.error('Create room error:', error);
    res.status(500).json({ error: 'Помилка створення номера' });
  }
};

export const updateRoomStatus = async (req: Request, res: Response) => {
