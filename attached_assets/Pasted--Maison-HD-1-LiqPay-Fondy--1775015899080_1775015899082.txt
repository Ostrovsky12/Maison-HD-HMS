// server/services/payments/liqpay.ts

import crypto from 'crypto';
import axios from 'axios';
import { db } from '../../db';
import { payments, subscriptions, invoices } from '../../db/schema';
import { eq } from 'drizzle-orm';

interface LiqPayConfig {
  publicKey: string;
  privateKey: string;
  apiUrl: string;
}

export class LiqPayService {
  private config: LiqPayConfig;
  
  constructor() {
    this.config = {
      publicKey: process.env.LIQPAY_PUBLIC_KEY!,
      privateKey: process.env.LIQPAY_PRIVATE_KEY!,
      apiUrl: process.env.LIQPAY_API_URL || 'https://www.liqpay.ua/api',
    };
  }
  
  // Створення платежу
  async createPayment(params: {
    organizationId: number;
    invoiceId: number;
    amount: number;
    description: string;
    successUrl: string;
    failureUrl: string;
  }) {
    const { organizationId, invoiceId, amount, description, successUrl, failureUrl } = params;
    
    const orderId = `INV-${invoiceId}-${Date.now()}`;
    
    const data = {
      version: '3',
      public_key: this.config.publicKey,
      action: 'pay',
      amount: amount.toString(),
      currency: 'UAH',
      description: description,
      order_id: orderId,
      result_url: successUrl,
      server_url: `${process.env.APP_URL}/api/webhooks/liqpay`,
      sandbox: process.env.NODE_ENV === 'development' ? '1' : '0',
    };
    
    const dataBase64 = Buffer.from(JSON.stringify(data)).toString('base64');
    const signature = this.generateSignature(dataBase64);
    
    // Зберігаємо інформацію про платіж
    await db.insert(payments).values({
      invoiceId,
      organizationId,
      paymentSystem: 'liqpay',
      paymentId: orderId,
      amount: amount.toString(),
      status: 'pending',
    });
    
    return {
      data: dataBase64,
      signature,
      formUrl: 'https://www.liqpay.ua/api/3/checkout',
    };
  }
  
  // Обробка webhook від LiqPay
  async handleWebhook(body: any) {
    const { data, signature } = body;
    
    // Перевірка підпису
    const expectedSignature = this.generateSignature(data);
    if (signature !== expectedSignature) {
      throw new Error('Invalid signature');
    }
    
    const decodedData = JSON.parse(Buffer.from(data, 'base64').toString());
    const { order_id, status, amount, payment_id, transaction_id } = decodedData;
    
    // Знаходимо платіж
    const [payment] = await db.select()
      .from(payments)
      .where(eq(payments.paymentId, order_id));
    
    if (!payment) {
      throw new Error('Payment not found');
    }
    
    // Оновлюємо статус платежу
    await db.update(payments)
      .set({
        status: status === 'success' ? 'success' : 'failure',
        rawResponse: decodedData,
        paidAt: status === 'success' ? new Date() : undefined,
      })
      .where(eq(payments.id, payment.id));
    
    // Якщо платіж успішний — активуємо підписку
    if (status === 'success') {
      await this.activateSubscription(payment.invoiceId!);
    }
    
    return { success: status === 'success', paymentId: transaction_id };
  }
  
  // Активація підписки після оплати
  private async activateSubscription(invoiceId: number) {
    const [invoice] = await db.select()
      .from(invoices)
      .where(eq(invoices.id, invoiceId));
    
    if (!invoice) return;
    
    // Оновлюємо статус рахунку
    await db.update(invoices)
      .set({ status: 'paid', paidAt: new Date() })
      .where(eq(invoices.id, invoiceId));
    
    // Оновлюємо або створюємо підписку
    const existing = await db.select()
      .from(subscriptions)
      .where(eq(subscriptions.organizationId, invoice.organizationId));
    
    const periodEnd = new Date();
    if (invoice.subscriptionPeriod === 'monthly') {
      periodEnd.setMonth(periodEnd.getMonth() + 1);
    } else {
      periodEnd.setFullYear(periodEnd.getFullYear() + 1);
    }
    
    if (existing.length > 0) {
      await db.update(subscriptions)
        .set({
          tier: invoice.subscriptionTier,
          period: invoice.subscriptionPeriod,
          status: 'active',
          currentPeriodStart: new Date(),
          currentPeriodEnd: periodEnd,
          lastPaymentId: invoiceId,
          updatedAt: new Date(),
        })
        .where(eq(subscriptions.organizationId, invoice.organizationId));
    } else {
      await db.insert(subscriptions).values({
        organizationId: invoice.organizationId,
        tier: invoice.subscriptionTier,
        period: invoice.subscriptionPeriod,
        status: 'active',
        currentPeriodStart: new Date(),
        currentPeriodEnd: periodEnd,
        lastPaymentId: invoiceId,
      });
    }
    
    // Відправляємо email підтвердження
    await this.sendPaymentConfirmation(invoice);
  }
  
  private generateSignature(data: string): string {
    const str = this.config.privateKey + data + this.config.privateKey;
    return crypto.createHash('sha1').update(str).digest('base64');
  }
  
  private async sendPaymentConfirmation(invoice: any) {
    // TODO: Відправка email
    console.log(`Payment confirmed for invoice ${invoice.invoiceNumber}`);
  }
}

import crypto from 'crypto';
import axios from 'axios';
import { db } from '../../db';
import { payments, subscriptions, invoices } from '../../db/schema';
import { eq } from 'drizzle-orm';

export class FondyService {
  private merchantId: string;
  private password: string;
  private apiUrl: string;
  
  constructor() {
    this.merchantId = process.env.FONDY_MERCHANT_ID!;
    this.password = process.env.FONDY_PASSWORD!;
    this.apiUrl = process.env.FONDY_API_URL || 'https://api.fondy.eu/api';
  }
  
  // Створення платежу
  async createPayment(params: {
    organizationId: number;
    invoiceId: number;
    amount: number;
    description: string;
    successUrl: string;
    failureUrl: string;
  }) {
    const { organizationId, invoiceId, amount, description, successUrl, failureUrl } = params;
    
    const orderId = `INV-${invoiceId}-${Date.now()}`;
    const orderDesc = description.substring(0, 100);
    
    const requestData = {
      merchant_id: this.merchantId,
      order_id: orderId,
      order_desc: orderDesc,
      amount: Math.round(amount * 100), // в копійках
      currency: 'UAH',
      response_url: successUrl,
      server_callback_url: `${process.env.APP_URL}/api/webhooks/fondy`,
      sender_email: params.organizationId.toString(),
      lang: 'ua',
    };
    
    const signature = this.generateSignature(requestData);
    
    const response = await axios.post(`${this.apiUrl}/checkout/url`, {
      request: {
        ...requestData,
        signature,
      },
    });
    
    if (response.data.response.response_status === 'success') {
      const paymentUrl = response.data.response.checkout_url;
      
      // Зберігаємо інформацію про платіж
      await db.insert(payments).values({
        invoiceId,
        organizationId,
        paymentSystem: 'fondy',
        paymentId: orderId,
        amount: amount.toString(),
        status: 'pending',
        rawResponse: response.data,
      });
      
      return { paymentUrl, orderId };
    }
    
    throw new Error('Failed to create payment');
  }
  
  // Обробка webhook від Fondy
  async handleWebhook(body: any) {
    const { response } = body;
    const { order_id, order_status, amount, payment_id } = response;
    
    // Перевірка підпису
    const expectedSignature = this.generateSignature(response);
    if (response.signature !== expectedSignature) {
      throw new Error('Invalid signature');
    }
    
    // Знаходимо платіж
    const [payment] = await db.select()
      .from(payments)
      .where(eq(payments.paymentId, order_id));
    
    if (!payment) {
      throw new Error('Payment not found');
    }
    
    const isSuccess = order_status === 'approved';
    
    // Оновлюємо статус платежу
    await db.update(payments)
      .set({
        status: isSuccess ? 'success' : 'failure',
        rawResponse: response,
        paidAt: isSuccess ? new Date() : undefined,
      })
      .where(eq(payments.id, payment.id));
    
    // Якщо платіж успішний — активуємо підписку
    if (isSuccess && payment.invoiceId) {
      await this.activateSubscription(payment.invoiceId);
    }
    
    return { success: isSuccess };
  }
  
  private generateSignature(data: any): string {
    const keys = Object.keys(data).sort();
    const str = keys.map(key => `${key}=${data[key]}`).join('|');
    return crypto.createHash('sha1').update(`${str}|${this.password}`).digest('hex');
  }

      .where(eq(invoices.id, invoiceId));

// server/services/invoicing.ts

import { db } from '../db';
import { invoices, subscriptions, organizations } from '../db/schema';
import { eq, and, lt } from 'drizzle-orm';
import PDFDocument from 'pdfkit';
import fs from 'fs';
import path from 'path';

export class InvoicingService {
  
  // Автоматичне створення рахунків для всіх активних підписок
  async generateMonthlyInvoices() {
    const activeSubscriptions = await db.select()
      .from(subscriptions)
      .where(
        and(
          eq(subscriptions.status, 'active'),
          lt(subscriptions.currentPeriodEnd, new Date())
        )
      );
    
    for (const sub of activeSubscriptions) {
      await this.createInvoice(sub);
      
      // Якщо автопоновлення вимкнено — деактивуємо
      if (!sub.autoRenew) {
        await db.update(subscriptions)
          .set({ status: 'cancelled' })
          .where(eq(subscriptions.id, sub.id));
      }
    }
  }
  
  // Створення рахунку для організації
  async createInvoice(subscription: any) {
    const [org] = await db.select()
      .from(organizations)
      .where(eq(organizations.id, subscription.organizationId));
    
    const prices: Record<string, number> = {
      basic: 1500,
      pro: 3500,
      enterprise: 0, // за запитом
    };
    
    const amount = prices[subscription.tier] || 0;
    const tax = amount * 0.2; // 20% ПДВ
    const total = amount + tax;
    
    const invoiceNumber = `INV-${new Date().getFullYear()}${String(new Date().getMonth() + 1).padStart(2, '0')}-${String(subscription.id).padStart(6, '0')}`;
    
    const [invoice] = await db.insert(invoices).values({
      organizationId: subscription.organizationId,
      invoiceNumber,
      invoiceDate: new Date(),
      dueDate: new Date(Date.now() + 14 * 24 * 60 * 60 * 1000),
      amount: amount.toString(),
      tax: tax.toString(),
      total: total.toString(),
      subscriptionTier: subscription.tier,
      subscriptionPeriod: subscription.period,
      periodStart: subscription.currentPeriodStart,
      periodEnd: subscription.currentPeriodEnd,
      status: 'pending',
    }).returning();
    
    // Генеруємо PDF
    const pdfUrl = await this.generatePDF(invoice, org);
    
    await db.update(invoices)
      .set({ pdfUrl })
      .where(eq(invoices.id, invoice.id));
    
    // Відправляємо email з рахунком
    await this.sendInvoiceEmail(org, invoice, pdfUrl);
    
    return invoice;
  }
  
  // Генерація PDF рахунку
  private async generatePDF(invoice: any, organization: any): Promise<string> {
    return new Promise((resolve, reject) => {
      const doc = new PDFDocument({ margin: 50 });

    .where(eq(invoices.id, parseInt(req.params.id)))
    .get();
  
  if (!invoice) {
    return res.status(404).json({ error: 'Invoice not found' });
  }
  
  res.json({ success: true, data: invoice });
});

  const invoice = await db.select()
    .from(invoices)
    .where(eq(invoices.id, parseInt(req.params.id)))
    .get();
  
  if (!invoice || invoice.status !== 'pending') {
    return res.status(400).json({ error: 'Invoice cannot be paid' });

  const liqpay = new LiqPayService();

    organizationId: req.user.organizationId,
    invoiceId: invoice.id,

    description: `Оплата підписки ${invoice.subscriptionTier} (${invoice.subscriptionPeriod})`,
    successUrl: `${process.env.APP_URL}/billing/success`,
    failureUrl: `${process.env.APP_URL}/billing/failure`,
  });
  
  res.json({ success: true, payment });
});
