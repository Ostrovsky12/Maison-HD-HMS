from datetime import datetime
from app import db

class Hotel(db.Model):
    """Модель готелю / об'єкта"""
    __tablename__ = 'hotels'
    
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(200), nullable=False)
    code = db.Column(db.String(20), unique=True, nullable=False)
    
    # Контактна інформація
    address = db.Column(db.Text)
    phone = db.Column(db.String(50))
    email = db.Column(db.String(100))
    website = db.Column(db.String(200))
    
    # Фінансові налаштування
    currency = db.Column(db.String(3), default='UAH')
    tax_rate = db.Column(db.Float, default=20.0)
    
    # Системні поля
    timezone = db.Column(db.String(50), default='Europe/Kiev')
    checkin_time = db.Column(db.String(5), default='14:00')
    checkout_time = db.Column(db.String(5), default='12:00')
    
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, onupdate=datetime.utcnow)
    is_active = db.Column(db.Boolean, default=True)
    
    # Зв'язки
    rooms = db.relationship('Room', backref='hotel', lazy='dynamic')
    users = db.relationship('User', backref='hotel', lazy='dynamic')
    room_categories = db.relationship('RoomCategory', backref='hotel', lazy='dynamic')
    
    def __repr__(self):
        return f'<Hotel {self.code}: {self.name}>'
    
    def to_dict(self):
        """Конвертація в словник для API"""
        return {
            'id': self.id,
            'name': self.name,
            'code': self.code,
            'address': self.address,
            'phone': self.phone,
            'email': self.email,
            'currency': self.currency,
            'tax_rate': self.tax_rate,

            'checkout_time': self.checkout_time,

"""
Authentication API
"""

from flask import request, jsonify
from flask_jwt_extended import create_access_token, jwt_required, get_jwt_identity
from datetime import timedelta

from app import db, bcrypt
from api import api_bp
from models.user import User


@api_bp.route('/auth/register', methods=['POST'])
def register():
    """Реєстрація нового користувача"""
    data = request.get_json()
    
    # Перевірка наявності користувача
    if User.query.filter_by(username=data.get('username')).first():
        return jsonify({'error': 'Username already exists'}), 400
    
    if User.query.filter_by(email=data.get('email')).first():
        return jsonify({'error': 'Email already exists'}), 400
    
    # Створення користувача
    user = User(
        username=data['username'],
        email=data['email'],
        full_name=data.get('full_name')
    )
    user.set_password(data['password'])
    
    db.session.add(user)
    db.session.commit()
    
    return jsonify({
        'message': 'User created successfully',
        'user': user.to_dict()
    }), 201


@api_bp.route('/auth/login', methods=['POST'])
def login():
    """Вхід в систему"""
    data = request.get_json()
    
    # Пошук користувача за username або email
    user = User.query.filter(
        (User.username == data.get('username')) | 
        (User.email == data.get('username'))
    ).first()
    
    if not user or not user.check_password(data.get('password')):
        return jsonify({'error': 'Invalid credentials'}), 401
    
    if not user.is_active:
        return jsonify({'error': 'Account is disabled'}), 401

@api_bp.route('/rooms/availability', methods=['GET'])
def check_availability():
    """
    Перевірка доступності номерів на задані дати
    GET /api/rooms/availability?hotel_id=1&check_in=2024-01-15&check_out=2024-01-17
    """
    hotel_id = request.args.get('hotel_id', type=int, required=True)
    check_in_str = request.args.get('check_in')
    check_out_str = request.args.get('check_out')
    
    if not check_in_str or not check_out_str:
        return jsonify({'error': 'check_in and check_out are required'}), 400
    
    try:
        check_in = datetime.strptime(check_in_str, '%Y-%m-%d')
        check_out = datetime.strptime(check_out_str, '%Y-%m-%d')
    except ValueError:
        return jsonify({'error': 'Invalid date format. Use YYYY-MM-DD'}), 400
    
    # Отримуємо всі категорії готелю
    categories = RoomCategory.query.filter_by(hotel_id=hotel_id, is_active=True).all()
    
    # Отримуємо всі бронювання на ці дати
    reservations = Reservation.query.filter(
        Reservation.hotel_id == hotel_id,
        Reservation.status.in_(['confirmed', 'checked_in']),
        Reservation.check_in < check_out,
        Reservation.check_out > check_in
    ).all()
    
    # Рахуємо зайняті номери по категоріях
    occupied_counts = {}
    for r in reservations:
        if r.room_id:
            room = Room.query.get(r.room_id)
            if room:
                cat_id = room.category_id
                occupied_counts[cat_id] = occupied_counts.get(cat_id, 0) + 1
    
    # Формуємо результат
    availability = []
    for category in categories:
        total_rooms = Room.query.filter_by(category_id=category.id).count()
        occupied = occupied_counts.get(category.id, 0)
        
        availability.append({
            'category_id': category.id,
            'category_name': category.name,
            'total_rooms': total_rooms,
            'occupied': occupied,
            'available': total_rooms - occupied,
            'base_price': category.base_price
        })
    
    return jsonify({
        'check_in': check_in_str,
        'check_out': check_out_str,
        'nights': (check_out - check_in).days,
        'availability': availability
    }), 200

    clean_rooms = Room.query.filter_by(hotel_id=hotel_id, status='clean').count()
    dirty_rooms = Room.query.filter_by(hotel_id=hotel_id, status='dirty').count()
    occupied_rooms = Room.query.filter_by(hotel_id=hotel_id, status='occupied').count()
    
    # Сьогоднішні заїзди/виїзди
    today = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
    tomorrow = today + timedelta(days=1)
    
    checkins_today = Reservation.query.filter(
        Reservation.hotel_id == hotel_id,
        Reservation.check_in >= today,
        Reservation.check_in < tomorrow,
        Reservation.status == 'confirmed'
    ).count()
    
    checkouts_today = Reservation.query.filter(
        Reservation.hotel_id == hotel_id,
        Reservation.check_out >= today,
        Reservation.check_out < tomorrow,
        Reservation.status == 'checked_in'
    ).count()
    
    # Заповненість
    occupancy_rate = (occupied_rooms / total_rooms * 100) if total_rooms > 0 else 0
    
    return jsonify({
        'total_rooms': total_rooms,
        'clean_rooms': clean_rooms,
        'dirty_rooms': dirty_rooms,
        'occupied_rooms': occupied_rooms,

// API Client
const API = {
    baseURL: '/api',
    
    async request(endpoint, options = {}) {
        const token = localStorage.getItem('access_token');
        const headers = {
            'Content-Type': 'application/json',
            ...options.headers
        };
        
        if (token) {
            headers['Authorization'] = `Bearer ${token}`;
        }
        
        try {
            const response = await fetch(`${this.baseURL}${endpoint}`, {
                ...options,
                headers
            });
            
            if (!response.ok) {
                const error = await response.json();
                throw new Error(error.error || 'API request failed');
            }
            
            return await response.json();
        } catch (error) {
            console.error('API Error:', error);
            throw error;
        }
    },
    
    get(endpoint) {
        return this.request(endpoint, { method: 'GET' });
    },
    
    post(endpoint, data) {
        return this.request(endpoint, {
            method: 'POST',
            body: JSON.stringify(data)
        });
    },
    
    put(endpoint, data) {
        return this.request(endpoint, {
            method: 'PUT',
            body: JSON.stringify(data)
        });
    }
};

function connectWebSocket() {

    const wsUrl = `${protocol}//${window.location.host}/ws`;
    
    ws = new WebSocket(wsUrl);

        console.log('WebSocket connected');
    };

        const data = JSON.parse(event.data);
        handleRealtimeUpdate(data);
    };
    
    ws.onclose = () => {
        console.log('WebSocket disconnected, reconnecting...');
        setTimeout(connectWebSocket, 5000);
