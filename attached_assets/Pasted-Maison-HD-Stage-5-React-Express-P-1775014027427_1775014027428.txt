export const kpiHistory = pgTable('kpi_history', {
  id: serial('id').primaryKey(),
  propertyId: integer('property_id').references(() => properties.id).notNull(),
  date: date('date').notNull(),
  
  // Основні метрики
  occupancyRate: decimal('occupancy_rate', { precision: 5, scale: 2 }).notNull(),
  adr: decimal('adr', { precision: 10, scale: 2 }).notNull(),      // Average Daily Rate
  revpar: decimal('revpar', { precision: 10, scale: 2 }).notNull(), // Revenue Per Available Room
  totalRevenue: decimal('total_revenue', { precision: 12, scale: 2 }).notNull(),
  
  // Додаткові
  checkins: integer('checkins').default(0),
  checkouts: integer('checkouts').default(0),
  avgStay: decimal('avg_stay', { precision: 4, scale: 1 }).default('0'),
  cancellationRate: decimal('cancellation_rate', { precision: 5, scale: 2 }).default('0'),
  
  // Канали продажів
  directBookings: integer('direct_bookings').default(0),
  otaBookings: integer('ota_bookings').default(0),

  otaRevenue: decimal('ota_revenue', { precision: 12, scale: 2 }).default('0'),

import { Router } from 'express';

const router = Router();

// GET /api/analytics/kpi?propertyId=1&days=60
router.get('/kpi', authenticate, async (req, res) => {
  const propertyId = parseInt(req.query.propertyId as string);
  const days = parseInt(req.query.days as string) || 60;
  
  const startDate = new Date();
  startDate.setDate(startDate.getDate() - days);
  
  const history = await db.select()
    .from(kpiHistory)
    .where(and(
      eq(kpiHistory.propertyId, propertyId),
      between(kpiHistory.date, startDate, new Date())
    ))
    .orderBy(kpiHistory.date);
  
  res.json({
    success: true,
    data: history.map(h => ({
      ...h,
      occupancyRate: parseFloat(h.occupancyRate),
      adr: parseFloat(h.adr),
      revpar: parseFloat(h.revpar),
      totalRevenue: parseFloat(h.totalRevenue),
    })),
  });
});

// GET /api/analytics/forecast?propertyId=1&days=30
router.get('/forecast', authenticate, async (req, res) => {
  const propertyId = parseInt(req.query.propertyId as string);
  const days = parseInt(req.query.days as string) || 30;
  
  const startDate = new Date();
  const endDate = new Date();
  endDate.setDate(endDate.getDate() + days);

    success: true,
    data: forecastData.map(f => ({
      ...f,
      predictedOccupancy: parseFloat(f.predictedOccupancy),

      predictedAdr: parseFloat(f.predictedAdr),
    })),
  });

  const year = parseInt(req.query.year as string);
  const month = parseInt(req.query.month as string);

    .where(and(
      eq(kpiHistory.propertyId, propertyId),

    .orderBy(kpiHistory.date);

    totalRevenue: data.reduce((sum, d) => sum + parseFloat(d.totalRevenue), 0),
    avgOccupancy: data.reduce((sum, d) => sum + parseFloat(d.occupancyRate), 0) / (data.length || 1),
    avgAdr: data.reduce((sum, d) => sum + parseFloat(d.adr), 0) / (data.length || 1),
    totalCheckins: data.reduce((sum, d) => sum + d.checkins, 0),
    totalCheckouts: data.reduce((sum, d) => sum + d.checkouts, 0),
  };
  
  res.json({ success: true, summary, daily: data });
});

import { db } from '../db';
import { kpiHistory, forecasts, bookings, properties } from '../db/schema';
import { eq, and, between, desc, sql } from 'drizzle-orm';

interface HistoryPoint {
  date: Date;
  occupancy: number;
  adr: number;
  revenue: number;
  weekday: number;
  month: number;
}

export class ForecastingService {
  constructor(private propertyId: number) {}
  
  async generateForecast(days: number = 90): Promise<any[]> {
    const history = await this.getHistory();
    
    if (history.length < 30) {
      return this.getFallbackForecast(days);
    }
    
    const forecasts = [];
    const startDate = new Date();
    
    for (let i = 0; i < days; i++) {
      const date = new Date(startDate);
      date.setDate(date.getDate() + i);
      
      const predictedOccupancy = this.predictOccupancy(history, date);
      const predictedAdr = this.predictAdr(history, date);
      
      const totalRooms = await this.getTotalRooms();
      const predictedRevenue = (predictedOccupancy / 100) * totalRooms * predictedAdr;
      
      const confidence = this.calculateConfidence(history, predictedOccupancy);
      
      const forecast = {
        propertyId: this.propertyId,
        forecastDate: date,
        predictedOccupancy,
        predictedRevenue,
        predictedAdr,
        confidenceLower: confidence.lower,
        confidenceUpper: confidence.upper,
        seasonFactor: this.getSeasonFactor(date),
        eventsFactor: 1.0,
      };
      
      forecasts.push(forecast);
    }
    
    // Зберігаємо в базу
    for (const f of forecasts) {
      await db.insert(forecasts).values(f).onConflictDoUpdate({
        target: [forecasts.propertyId, forecasts.forecastDate],
        set: f,
      });
    }
    
    return forecasts;
  }
  
  private async getHistory(): Promise<HistoryPoint[]> {
    const startDate = new Date();
    startDate.setFullYear(startDate.getFullYear() - 1);
    
    const data = await db.select()
      .from(kpiHistory)
      .where(and(
        eq(kpiHistory.propertyId, this.propertyId),
        between(kpiHistory.date, startDate, new Date())
      ))
      .orderBy(kpiHistory.date);
    
    return data.map(d => ({
      date: d.date,
      occupancy: parseFloat(d.occupancyRate),
      adr: parseFloat(d.adr),
      revenue: parseFloat(d.totalRevenue),
      weekday: d.date.getDay(),
      month: d.date.getMonth() + 1,
    }));
  }
  
  private predictOccupancy(history: HistoryPoint[], date: Date): number {
    const weekday = date.getDay();
    const month = date.getMonth() + 1;
    const isWeekend = weekday === 0 || weekday === 6;
    const seasonFactor = this.getSeasonFactor(date);
    
    // Фільтруємо схожі дні з історії
    const similarDays = history.filter(h => 
      h.weekday === weekday && 
      Math.abs(h.month - month) <= 1
    );
    
    if (similarDays.length === 0) {
      const avgOccupancy = history.reduce((sum, h) => sum + h.occupancy, 0) / history.length;
      return Math.min(100, avgOccupancy * seasonFactor);
    }
    
    const avgSimilar = similarDays.reduce((sum, h) => sum + h.occupancy, 0) / similarDays.length;
    let predicted = avgSimilar * seasonFactor;
    
    // Коригування для вихідних
    if (isWeekend) predicted *= 1.15;
    
    return Math.min(100, Math.max(0, predicted));
  }
  
  private predictAdr(history: HistoryPoint[], date: Date): number {
    const recent = history.slice(0, 60);
    const avgAdr = recent.reduce((sum, h) => sum + h.adr, 0) / recent.length;
    
    const seasonFactor = this.getSeasonFactor(date);
    const isWeekend = date.getDay() === 0 || date.getDay() === 6;
    const weekdayFactor = isWeekend ? 1.2 : 1.0;
    
    return avgAdr * seasonFactor * weekdayFactor;
  }
  
  private calculateConfidence(history: HistoryPoint[], prediction: number): { lower: number; upper: number } {
    const occupancies = history.slice(0, 30).map(h => h.occupancy);
    const stdDev = this.calculateStdDev(occupancies);
    const margin = stdDev * 1.96; // 95% довірчий інтервал
    
    return {
      lower: Math.max(0, prediction - margin),
      upper: Math.min(100, prediction + margin),
    };
  }
  
  private getSeasonFactor(date: Date): number {
    const month = date.getMonth() + 1;
    if (month >= 6 && month <= 8) return 1.3;      // Літо
    if (month === 5 || month === 9) return 1.1;    // Травень/Вересень
    if (month >= 11 || month <= 3) return 0.85;    // Зима
    return 1.0;
  }
  
  private async getTotalRooms(): Promise<number> {
    const result = await db.select({ count: sql<number>`count(*)` })
      .from(properties)
      .where(eq(properties.id, this.propertyId));
    
    return result[0]?.count || 0;
  }
  
  private calculateStdDev(values: number[]): number {
    const mean = values.reduce((a, b) => a + b, 0) / values.length;
    const variance = values.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / values.length;
    return Math.sqrt(variance);
  }
  
  private getFallbackForecast(days: number): any[] {
    const forecasts = [];
    const startDate = new Date();
    
    for (let i = 0; i < days; i++) {
      const date = new Date(startDate);
      date.setDate(date.getDate() + i);
      
      forecasts.push({
        propertyId: this.propertyId,
        forecastDate: date,
        predictedOccupancy: 65,
        predictedRevenue: 35000,
        predictedAdr: 1800,
        confidenceLower: 45,
        confidenceUpper: 85,
      });
    }
    
    return forecasts;
  }
}

import axios from 'axios';
import { db } from '../db';
import { bookings, guests, units, otaChannels } from '../db/schema';
import { eq, and, gt } from 'drizzle-orm';

interface OTAChannel {
  id: number;
  name: string;
  channelType: string;
  apiKey: string;
  apiSecret: string;
  apiUrl: string;
  propertyId: number;
}

export class OTASyncService {
  constructor(private channel: OTAChannel) {}
  
  async sync(): Promise<{ success: boolean; data?: any; error?: string }> {
    try {
      const newBookings = await this.fetchBookings();
      const imported = [];
      
      for (const booking of newBookings) {
        const importedBooking = await this.importBooking(booking);
        imported.push(importedBooking);
      }
      
      await this.updateAvailability();
      
      return { success: true, data: { imported, count: imported.length } };
    } catch (error: any) {
      return { success: false, error: error.message };
    }
  }
  
  private async fetchBookings(): Promise<any[]> {
    const lastSync = this.channel.lastSync;
    
    const response = await axios.get(`${this.channel.apiUrl}/bookings`, {
      headers: {
        'Authorization': `Bearer ${this.channel.apiKey}`,
        'Content-Type': 'application/json',
      },
      params: lastSync ? { last_update: lastSync.toISOString() } : {},
      timeout: 30000,
    });
    
    return response.data.bookings || [];
  }
  
  private async importBooking(bookingData: any): Promise<any> {
    // Пошук або створення гостя
    let guest = await db.select().from(guests).where(eq(guests.email, bookingData.guest_email)).get();
    
    if (!guest) {
      [guest] = await db.insert(guests).values({
        firstName: bookingData.guest_first_name,
        lastName: bookingData.guest_last_name,
        email: bookingData.guest_email,
        phone: bookingData.guest_phone,
      }).returning();
    }
    
    // Пошук вільного номера
    const checkIn = new Date(bookingData.check_in);
    const checkOut = new Date(bookingData.check_out);
    
    const availableUnits = await this.findAvailableUnits(checkIn, checkOut);
    const unit = availableUnits[0];
    
    // Створення бронювання
    const [booking] = await db.insert(bookings).values({
      propertyId: this.channel.propertyId,
      guestId: guest.id,
      unitId: unit?.id,
      checkIn,
      checkOut,
      adults: bookingData.adults,
      children: bookingData.children,
      totalAmount: bookingData.total_amount,
      source: this.channel.channelType,
      sourceId: bookingData.booking_id,
      status: 'confirmed',
    }).returning();
    
    // Оновлення статистики каналу
    await db.update(otaChannels)
      .set({
        totalBookings: this.channel.totalBookings + 1,
        totalRevenue: (this.channel.totalRevenue + booking.totalAmount).toString(),
      })
      .where(eq(otaChannels.id, this.channel.id));
    
    return booking;
  }
  
  private async findAvailableUnits(checkIn: Date, checkOut: Date): Promise<any[]> {
    const occupied = await db.select({ unitId: bookings.unitId })
      .from(bookings)
      .where(and(
        eq(bookings.propertyId, this.channel.propertyId),
        gt(bookings.checkOut, checkIn),
        gt(checkOut, bookings.checkIn)
      ));
    
    const occupiedIds = occupied.map(o => o.unitId);
    
    const available = await db.select()
      .from(units)
      .where(and(
        eq(units.propertyId, this.channel.propertyId),
        eq(units.isActive, true),
        occupiedIds.length ? sql`${units.id} NOT IN (${occupiedIds.join(',')})` : sql`1=1`
      ))
      .limit(1);
    
    return available;
  }
  
  private async updateAvailability(): Promise<void> {
    // Оновлення доступності в OTA на найближчі 90 днів
    const availability = await this.calculateAvailability();
    
    await axios.post(`${this.channel.apiUrl}/availability`, availability, {
      headers: {
        'Authorization': `Bearer ${this.channel.apiKey}`,
        'Content-Type': 'application/json',
      },
    });
  }
  
  private async calculateAvailability(): Promise<any> {
    // Розрахунок доступності на найближчі 90 днів
    const unitsCount = await db.select({ count: sql<number>`count(*)` })
      .from(units)
      .where(eq(units.propertyId, this.channel.propertyId));
    
    const availability = {};
    const startDate = new Date();
    
    for (let i = 0; i < 90; i++) {
      const date = new Date(startDate);
      date.setDate(date.getDate() + i);
      
      const occupied = await db.select({ count: sql<number>`count(*)` })
        .from(bookings)
        .where(and(
          eq(bookings.propertyId, this.channel.propertyId),
          sql`${bookings.checkIn} <= ${date}`,
          sql`${bookings.checkOut} > ${date}`,
          sql`${bookings.status} IN ('confirmed', 'checked_in')`
        ));
      
      availability[date.toISOString().split('T')[0]] = {
        available: unitsCount[0].count - occupied[0].count,
      };
    }
    
    return { availability };
  }
}

import { useState } from 'react';

import { useParams } from 'wouter';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';

import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';

import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { api } from '@/lib/api';
import { toast } from 'sonner';

interface OTAChannel {
  id: number;
  name: string;
  channelType: string;
  commissionRate: string;
  syncEnabled: boolean;
  lastSync: string;
  totalBookings: number;
  totalRevenue: string;
}

export function OTAChannels() {
  const { propertyId } = useParams();
  const queryClient = useQueryClient();
  const [isOpen, setIsOpen] = useState(false);
  
  const { data, isLoading } = useQuery({
    queryKey: ['ota-channels', propertyId],
    queryFn: () => api.get(`/integrations/ota?propertyId=${propertyId}`),
  });
  
  const syncMutation = useMutation({
    mutationFn: (channelId: number) => api.post(`/integrations/ota/${channelId}/sync`),
    onSuccess: () => {
      toast.success('Синхронізацію завершено');

  });
  
  const channels: OTAChannel[] = data?.data || [];

                  <TableCell className="font-medium">{channel.name}</TableCell>
                  <TableCell>
                    <span className="px-2 py-1 bg-blue-100 text-blue-700 rounded-full text-xs">
                      {channel.channelType}
                    </span>
                  </TableCell>
                  <TableCell>{channel.commissionRate}%</TableCell>
                  <TableCell>
                    <Switch 
                      checked={channel.syncEnabled}
                      onCheckedChange={() => {}}
                    />
                  </TableCell>
                  <TableCell className="text-sm text-gray-500">

                  </TableCell>

                  <TableCell>{parseFloat(channel.totalRevenue).toLocaleString()} ₴</TableCell>
                  <TableCell>

                      size="sm"

                  </TableCell>
                </TableRow>
              )}
            </TableBody>
          </Table>
        </CardContent>
      </Card>
    </div>

function AddChannelForm({ onSuccess, propertyId }: { onSuccess: () => void; propertyId: string }) {
  const [formData, setFormData] = useState({
    name: '',
    channelType: 'booking',
    apiKey: '',
    apiSecret: '',
    commissionRate: '15.0',
  });
  
  const mutation = useMutation({
    mutationFn: () => api.post('/integrations/ota', { ...formData, propertyId: parseInt(propertyId) }),
    onSuccess: () => {
      toast.success('Канал додано');
      onSuccess();
    },
    onError: () => toast.error('Помилка додавання каналу'),
  });
  
  return (
    <form onSubmit={(e) => { e.preventDefault(); mutation.mutate(); }} className="space-y-4">
      <div>
        <Label>Назва каналу</Label>
        <Input 
          value={formData.name}
          onChange={(e) => setFormData({ ...formData, name: e.target.value })}
          placeholder="Booking.com"
          required
        />
      </div>
      
      <div>
        <Label>Тип каналу</Label>
        <Select value={formData.channelType} onValueChange={(v) => setFormData({ ...formData, channelType: v })}>
          <SelectTrigger>
            <SelectValue />
          </SelectTrigger>
          <SelectContent>
            <SelectItem value="booking">Booking.com</SelectItem>
            <SelectItem value="expedia">Expedia</SelectItem>
            <SelectItem value="airbnb">Airbnb</SelectItem>
            <SelectItem value="agoda">Agoda</SelectItem>
          </SelectContent>
        </Select>
      </div>
      
      <div>
        <Label>API Key</Label>
        <Input 
          value={formData.apiKey}
          onChange={(e) => setFormData({ ...formData, apiKey: e.target.value })}
          type="password"
        />
      </div>
      
      <div>
        <Label>API Secret</Label>
        <Input 
          value={formData.apiSecret}
          onChange={(e) => setFormData({ ...formData, apiSecret: e.target.value })}
          type="password"
        />
      </div>
      
      <div>
        <Label>Комісія (%)</Label>
        <Input 
          type="number"
          step="0.1"
          value={formData.commissionRate}
          onChange={(e) => setFormData({ ...formData, commissionRate: e.target.value })}
        />
      </div>
      
      <div className="flex justify-end space-x-3 pt-4">
        <Button type="button" variant="outline" onClick={onSuccess}>Скасувати</Button>
        <Button type="submit" disabled={mutation.isPending}>
          {mutation.isPending ? 'Додається...' : 'Додати канал'}
        </Button>
      </div>
    </form>
  );
