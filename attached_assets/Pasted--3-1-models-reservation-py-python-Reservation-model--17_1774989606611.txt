"""
Reservation model - бронювання номерів з повною логікою
"""

from datetime import datetime, timedelta
import secrets
import string
from app import db

class Reservation(db.Model):
    """Модель бронювання"""
    __tablename__ = 'reservations'
    
    id = db.Column(db.Integer, primary_key=True)
    reservation_number = db.Column(db.String(20), unique=True, nullable=False)
    
    # Статуси
    status = db.Column(db.String(20), default='confirmed')
    # confirmed, checked_in, checked_out, cancelled, no_show
    
    # Дати
    check_in = db.Column(db.DateTime, nullable=False)
    check_out = db.Column(db.DateTime, nullable=False)
    actual_check_in = db.Column(db.DateTime)
    actual_check_out = db.Column(db.DateTime)
    
    # Гості
    guest_id = db.Column(db.Integer, db.ForeignKey('guests.id'), nullable=False)
    adults = db.Column(db.Integer, default=1)
    children = db.Column(db.Integer, default=0)
    extra_beds = db.Column(db.Integer, default=0)
    
    # Номери
    room_id = db.Column(db.Integer, db.ForeignKey('rooms.id'))
    room_category_id = db.Column(db.Integer, db.ForeignKey('room_categories.id'), nullable=False)
    
    # Фінанси
    total_amount = db.Column(db.Float, default=0)
    paid_amount = db.Column(db.Float, default=0)
    deposit_amount = db.Column(db.Float, default=0)
    discount = db.Column(db.Float, default=0)
    discount_type = db.Column(db.String(20), default='fixed')  # fixed, percent
    tax = db.Column(db.Float, default=0)
    
    # Джерело
    source = db.Column(db.String(50), default='direct')
    source_id = db.Column(db.String(100))
    
    # Додатково
    special_requests = db.Column(db.Text)
    notes = db.Column(db.Text)
    
    # Системні поля
    created_by = db.Column(db.Integer, db.ForeignKey('users.id'))
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, onupdate=datetime.utcnow)
    
    # Зв'язки
    room = db.relationship('Room', backref='reservations')
    room_category = db.relationship('RoomCategory', backref='reservations')
    guest = db.relationship('Guest', backref='reservations')
    creator = db.relationship('User', backref='created_reservations')
    invoices = db.relationship('Invoice', backref='reservation', lazy='dynamic')
    
    @staticmethod
    def generate_number():
        """Генерація унікального номера бронювання"""
        prefix = datetime.now().strftime('%Y%m%d')
        suffix = ''.join(secrets.choice(string.digits) for _ in range(6))
        return f"{prefix}-{suffix}"
    
    @property
    def nights(self):
        """Кількість ночей"""
        return (self.check_out - self.check_in).days
    
    @property
    def balance(self):
        """Залишок до оплати"""
        return self.total_amount - self.paid_amount
    
    @property
    def is_deposit_required(self):
        """Чи потрібен депозит"""
        return self.deposit_amount > 0 and self.paid_amount < self.deposit_amount
    
    @property
    def status_display(self):
        """Відображення статусу українською"""
        status_map = {
            'confirmed': 'Підтверджено',
            'checked_in': 'Заселено',
            'checked_out': 'Виселено',
            'cancelled': 'Скасовано',
            'no_show': 'Не з\'явився'
        }
        return status_map.get(self.status, self.status)
    
    def calculate_total(self):
        """Розрахунок загальної суми з урахуванням знижки"""
        from services.pricing import calculate_reservation_price
        
        price_data = calculate_reservation_price(
            category_id=self.room_category_id,
            check_in=self.check_in,
            check_out=self.check_out,
            adults=self.adults,
            extra_beds=self.extra_beds
        )
        
        subtotal = price_data['total']
        
        # Застосування знижки
        if self.discount > 0:
            if self.discount_type == 'percent':
                discount_amount = subtotal * (self.discount / 100)
            else:
                discount_amount = self.discount
            
            self.total_amount = subtotal - discount_amount
        else:
            self.total_amount = subtotal
        
        # Розрахунок ПДВ
        self.tax = self.total_amount * 0.2  # 20% ПДВ
        
        return self.total_amount
    
    def check_in_guest(self):
        """Заселення гостя"""
        if self.status != 'confirmed':
            raise ValueError(f"Cannot check in reservation with status {self.status}")
        
        self.status = 'checked_in'
        self.actual_check_in = datetime.utcnow()
        
        if self.room:
            self.room.status = 'occupied'
        
        db.session.commit()
        
        # Відправка сповіщення
        from services.telegram_bot import send_notification
        send_notification(f"🏨 Заселення: {self.guest.full_name} в номер {self.room.number if self.room else 'не призначено'}")
    
    def check_out_guest(self):
        """Виселення гостя"""
        if self.status != 'checked_in':
            raise ValueError(f"Cannot check out reservation with status {self.status}")
        
        self.status = 'checked_out'
        self.actual_check_out = datetime.utcnow()
        
        if self.room:
            self.room.status = 'dirty'
        
        # Оновлення статистики гостя
        if self.guest:
            self.guest.add_loyalty_points(
                points=self.total_amount * 0.05,  # 5% бонусів
                nights=self.nights,
                amount=self.total_amount
            )
        
        db.session.commit()
        
        # Відправка сповіщення
        from services.telegram_bot import send_notification
        send_notification(f"🚪 Виселення: {self.guest.full_name} з номеру {self.room.number if self.room else 'не призначено'}")
    
    def cancel(self, reason=None):
        """Скасування бронювання"""
        if self.status in ['checked_in', 'checked_out']:
            raise ValueError(f"Cannot cancel reservation with status {self.status}")
        
        self.status = 'cancelled'
        self.notes = (self.notes or '') + f"\nСкасовано: {reason}" if reason else ''
        
        db.session.commit()
        
        # Відправка сповіщення
        from services.telegram_bot import send_notification
        send_notification(f"❌ Скасовано бронювання #{self.reservation_number} для {self.guest.full_name}")
    
    def to_dict(self, detailed=False):
        result = {
            'id': self.id,
            'reservation_number': self.reservation_number,
            'status': self.status,
            'status_display': self.status_display,
            'check_in': self.check_in.isoformat(),
            'check_out': self.check_out.isoformat(),
            'actual_check_in': self.actual_check_in.isoformat() if self.actual_check_in else None,
            'actual_check_out': self.actual_check_out.isoformat() if self.actual_check_out else None,
            'nights': self.nights,
            'adults': self.adults,
            'children': self.children,
            'extra_beds': self.extra_beds,
            'total_amount': self.total_amount,
            'paid_amount': self.paid_amount,
            'balance': self.balance,
            'deposit_amount': self.deposit_amount,
            'discount': self.discount,
            'discount_type': self.discount_type,
            'source': self.source,
            'special_requests': self.special_requests,
            'notes': self.notes,
            'created_at': self.created_at.isoformat()
        }
        
        if detailed:
            result['guest'] = self.guest.to_dict() if self.guest else None
            result['room'] = self.room.to_dict() if self.room else None
            result['room_category'] = self.room_category.to_dict() if self.room_category else None
        
        return result

"""
Dynamic pricing engine
"""

from datetime import datetime, timedelta
from models.room import RoomCategory

def calculate_reservation_price(category_id, check_in, check_out, adults=1, extra_beds=0):
    """
    Розрахунок вартості бронювання з урахуванням:
    - Базової ціни категорії
    - Сезонності
    - Дня тижня
    - Заповненості
    - Кількості гостей
    """
    category = RoomCategory.query.get(category_id)
    if not category:
        raise ValueError(f"Category {category_id} not found")
    
    nights = (check_out - check_in).days
    daily_prices = []
    total = 0
    
    current_date = check_in
    while current_date < check_out:
        price = get_daily_price(category, current_date)
        daily_prices.append({
            'date': current_date.strftime('%Y-%m-%d'),
            'price': price
        })
        total += price
        current_date += timedelta(days=1)
    
    # Додаткова плата за додаткових гостей
    extra_guest_charge = 0
    if adults > category.capacity:
        extra = adults - category.capacity
        extra_guest_charge = extra * 500  # 500 грн за додаткового дорослого
    
    # Додаткові місця
    extra_bed_charge = extra_beds * category.extra_bed_price
    
    final_total = total + extra_guest_charge + extra_bed_charge
    
    return {
        'category': category.name,
        'nights': nights,
        'daily_prices': daily_prices,
        'subtotal': total,
        'extra_guest_charge': extra_guest_charge,
        'extra_bed_charge': extra_bed_charge,
        'total': final_total
    }

def get_daily_price(category, date):
    """
    Розрахунок ціни на конкретну дату з динамічними факторами
    """
    base_price = category.base_price
    
    # Фактор сезонності
    season_factor = get_season_factor(date)
    
    # Фактор дня тижня (вихідні дорожче)
    weekday_factor = 1.2 if date.weekday() >= 4 else 1.0
    
    # Фактор заповненості (динамічний, буде додано пізніше)
    occupancy_factor = 1.0
    
    # Фактор попереднього бронювання (раннє бронювання дешевше)
    days_until = (date - datetime.now().date()).days
    early_booking_factor = 0.9 if days_until > 30 else 1.0
    
    price = base_price * season_factor * weekday_factor * occupancy_factor * early_booking_factor
    
    return round(price, 2)

def get_season_factor(date):
    """
    Визначення сезонного коефіцієнта
    """
    month = date.month
    
    # Високий сезон (червень-серпень)
    if month in [6, 7, 8]:
        return 1.3
    # Середній сезон (травень, вересень)
    elif month in [5, 9]:
        return 1.1
    # Низький сезон (листопад-березень)
    elif month in [11, 12, 1, 2, 3]:
        return 0.85
    # Інше
    else:
        return 1.0

"""
Availability calculation service
"""

from datetime import datetime, timedelta
from models.room import Room, RoomCategory
from models.reservation import Reservation

def check_availability(hotel_id, check_in, check_out, category_id=None):
    """
    Перевірка доступності номерів на задані дати
    """
    # Отримуємо всі бронювання на ці дати
    reservations = Reservation.query.filter(
        Reservation.hotel_id == hotel_id,
        Reservation.status.in_(['confirmed', 'checked_in']),
        Reservation.check_in < check_out,
        Reservation.check_out > check_in
    ).all()
    
    # Рахуємо зайняті номери
    occupied_room_ids = set()
    for r in reservations:
        if r.room_id:
            occupied_room_ids.add(r.room_id)
    
    # Отримуємо всі номери
    query = Room.query.filter_by(hotel_id=hotel_id)
    if category_id:
        query = query.filter_by(category_id=category_id)
    
    all_rooms = query.all()
    
    available_rooms = [r for r in all_rooms if r.id not in occupied_room_ids]
    
    return {
        'total_rooms': len(all_rooms),
        'occupied': len(occupied_room_ids),
        'available': len(available_rooms),
        'available_rooms': [r.to_dict() for r in available_rooms]
    }

def get_occupancy_calendar(hotel_id, start_date, days=30):
    """
    Отримання календаря заповненості на N днів
    """
    calendar = []
    
    current_date = start_date
    for _ in range(days):
        next_date = current_date + timedelta(days=1)
        
        # Кількість зайнятих номерів на цю дату
        occupied = Reservation.query.filter(
            Reservation.hotel_id == hotel_id,
            Reservation.status.in_(['confirmed', 'checked_in']),
            Reservation.check_in < next_date,
            Reservation.check_out > current_date
        ).count()
        
        total_rooms = Room.query.filter_by(hotel_id=hotel_id).count()
        
        occupancy_rate = (occupied / total_rooms * 100) if total_rooms > 0 else 0
        
        calendar.append({
            'date': current_date.strftime('%Y-%m-%d'),
            'weekday': current_date.strftime('%A'),
            'occupied': occupied,
            'total': total_rooms,
            'occupancy_rate': round(occupancy_rate, 1)
        })
        
        current_date = next_date
    
    return calendar

"""
Telegram bot for notifications
"""

import os
import requests
from loguru import logger

TELEGRAM_TOKEN = os.getenv('TELEGRAM_BOT_TOKEN')
TELEGRAM_CHAT_ID = os.getenv('TELEGRAM_CHAT_ID')

def send_notification(message, parse_mode='HTML'):
    """
    Відправка сповіщення в Telegram
    """
    if not TELEGRAM_TOKEN or not TELEGRAM_CHAT_ID:
        logger.warning("Telegram credentials not configured")
        return False
    
    try:
        url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
        payload = {
            'chat_id': TELEGRAM_CHAT_ID,
            'text': message,
            'parse_mode': parse_mode
        }
        
        response = requests.post(url, json=payload, timeout=10)
        
        if response.status_code == 200:
            logger.info(f"Telegram notification sent: {message[:50]}...")
            return True
        else:
            logger.error(f"Failed to send Telegram notification: {response.text}")
            return False
            
    except Exception as e:
        logger.error(f"Telegram error: {e}")
        return False

def send_alert(alert_type, details):
    """
    Відправка термінового сповіщення про піковий момент
    """
    emoji_map = {
        'high_occupancy': '🔴',
        'checkin_wave': '🏨',
        'checkout_wave': '🚪',
        'low_availability': '⚠️',
        'vip_arrival': '👑',
        'system_alert': '🚨'
    }
    
    emoji = emoji_map.get(alert_type, '📢')
    message = f"{emoji} <b>Алерт: {alert_type}</b>\n\n{details}"
    
    return send_notification(message)

def check_peak_moments(hotel_id):
    """
    Перевірка пікових моментів для сповіщень
    Викликається періодично (кожні 15 хвилин)
    """
    from models.room import Room
    from models.reservation import Reservation
    from datetime import datetime, timedelta
    
    # 1. Висока заповненість (> 90%)
    total_rooms = Room.query.filter_by(hotel_id=hotel_id).count()
    occupied = Room.query.filter_by(hotel_id=hotel_id, status='occupied').count()
    occupancy_rate = (occupied / total_rooms * 100) if total_rooms > 0 else 0
    
    if occupancy_rate > 90:
        send_alert('high_occupancy', f"Заповненість досягла {occupancy_rate:.1f}%! Залишилось {total_rooms - occupied} вільних номерів.")
    
    # 2. Хвиля заїздів (більше 10 за день)
    today = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
    tomorrow = today + timedelta(days=1)
    
    checkins_today = Reservation.query.filter(
        Reservation.hotel_id == hotel_id,
        Reservation.check_in >= today,
        Reservation.check_in < tomorrow,
        Reservation.status == 'confirmed'
    ).count()
    
    if checkins_today > 10:
        send_alert('checkin_wave', f"Сьогодні очікується {checkins_today} заїздів! Підготуйте номери.")
    
    # 3. Хвиля виїздів (більше 10 за день)
    checkouts_today = Reservation.query.filter(
        Reservation.hotel_id == hotel_id,
        Reservation.check_out >= today,
        Reservation.check_out < tomorrow,
        Reservation.status == 'checked_in'
    ).count()
    
    if checkouts_today > 10:
        send_alert('checkout_wave', f"Сьогодні очікується {checkouts_today} виїздів! Підготуйте рахунки.")
    
    # 4. Низька доступність (< 5 номерів)
    available_rooms = Room.query.filter_by(hotel_id=hotel_id, status='clean').count()
    if available_rooms < 5 and occupancy_rate > 80:
        send_alert('low_availability', f"Залишилось лише {available_rooms} чистих номерів!")
    
    # 5. VIP гості сьогодні
    vip_guests = Reservation.query.join(Reservation.guest).filter(
        Reservation.hotel_id == hotel_id,
        Reservation.check_in >= today,
        Reservation.check_in < tomorrow,
        Reservation.guest.has(tags=['vip'])
    ).all()
    
    if vip_guests:
        names = ', '.join([g.guest.full_name for g in vip_guests])
        send_alert('vip_arrival', f"Сьогодні заселяються VIP гості: {names}")
    
    return True

from flask import request, jsonify
from flask_jwt_extended import jwt_required, get_jwt_identity

@api_bp.route('/reservations', methods=['GET'])
def get_reservations():
    """Отримання списку бронювань з фільтрами"""
    hotel_id = request.args.get('hotel_id', type=int, default=1)
    status = request.args.get('status')
    date_from = request.args.get('date_from')
    date_to = request.args.get('date_to')
    
    query = Reservation.query
    
    if status:
        query = query.filter_by(status=status)
    
    if date_from:
        query = query.filter(Reservation.check_in >= date_from)

@api_bp.route('/reservations/calendar', methods=['GET'])
def get_calendar():
    """Отримання календаря (шахматки)"""
    hotel_id = request.args.get('hotel_id', type=int, default=1)
    start_date_str = request.args.get('start_date')
    days = request.args.get('days', type=int, default=30)
    
    if not start_date_str:
        start_date = datetime.now().replace(day=1)
    else:
        start_date = datetime.strptime(start_date_str, '%Y-%m-%d')
    
    from services.availability import get_occupancy_calendar
    calendar = get_occupancy_calendar(hotel_id, start_date, days)
    
    # Отримуємо всі категорії
    categories = RoomCategory.query.filter_by(hotel_id=hotel_id, is_active=True).all()
    
    # Отримуємо бронювання для клітинок
    end_date = start_date + timedelta(days=days)
    reservations = Reservation.query.filter(
        Reservation.hotel_id == hotel_id,
        Reservation.status.in_(['confirmed', 'checked_in']),
        Reservation.check_in < end_date,
        Reservation.check_out > start_date
    ).all()
    
    # Групуємо бронювання по датах та категоріях
    grid = {}
    for r in reservations:
        current = max(r.check_in, start_date)
        while current < min(r.check_out, end_date):
            date_key = current.strftime('%Y-%m-%d')
            if date_key not in grid:
                grid[date_key] = {}
            
            cat_id = r.room_category_id
            if cat_id not in grid[date_key]:
                grid[date_key][cat_id] = 0
            grid[date_key][cat_id] += 1
            
            current += timedelta(days=1)
    
    return jsonify({
        'calendar': calendar,
        'categories': [c.to_dict() for c in categories],
        'grid': grid,
        'start_date': start_date.strftime('%Y-%m-%d'),
        'days': days
    }), 200

def create_reservation():
    """Створення нового бронювання"""
    data = request.get_json()
    
    # Перевірка обов'язкових полів
    required = ['guest_id', 'check_in', 'check_out', 'room_category_id']
    for field in required:
        if field not in data:
            return jsonify({'error': f'Missing field: {field}'}), 400
    
    # Парсинг дат
    try:
        check_in = datetime.fromisoformat(data['check_in'])
        check_out = datetime.fromisoformat(data['check_out'])
    except ValueError:
        return jsonify({'error': 'Invalid date format. Use ISO format'}), 400
    
    if check_in >= check_out:
        return jsonify({'error': 'Check-in must be before check-out'}), 400
    
    # Перевірка доступності
    availability = check_availability(
        hotel_id=1,  # TODO: з контексту користувача
        check_in=check_in,
        check_out=check_out,
        category_id=data['room_category_id']
    )
    
    if availability['available'] == 0:
        return jsonify({'error': 'No rooms available for these dates'}), 409
    
    # Отримання гостя
    guest = Guest.query.get(data['guest_id'])
    if not guest:
        return jsonify({'error': 'Guest not found'}), 404
    
    # Створення бронювання
    reservation = Reservation(
        reservation_number=Reservation.generate_number(),
        guest_id=guest.id,
        check_in=check_in,
        check_out=check_out,
        room_category_id=data['room_category_id'],
        adults=data.get('adults', 1),
        children=data.get('children', 0),
        extra_beds=data.get('extra_beds', 0),
        special_requests=data.get('special_requests'),
        source=data.get('source', 'direct'),
        created_by=get_jwt_identity()
    )
    
    # Розрахунок вартості
    price_data = calculate_reservation_price(
        category_id=data['room_category_id'],
        check_in=check_in,
        check_out=check_out,
        adults=reservation.adults,
        extra_beds=reservation.extra_beds
    )
    
    reservation.total_amount = price_data['total']
    
    # Якщо вказано конкретний номер
    if data.get('room_id'):
        room = Room.query.get(data['room_id'])
        if room and room.category_id == data['room_category_id']:
            reservation.room_id = room.id
    
    db.session.add(reservation)
    db.session.commit()
    
    # Відправка сповіщення
    send_notification(
        f"🆕 <b>Нове бронювання #{reservation.reservation_number}</b>\n\n"
        f"Гість: {guest.full_name}\n"
        f"Дати: {check_in.strftime('%d.%m.%Y')} - {check_out.strftime('%d.%m.%Y')}\n"
        f"Сума: {reservation.total_amount:,.2f} ₴"
    )
    
    return jsonify(reservation.to_dict(detailed=True)), 201

function updateRecentReservations(reservations) {
    const container = document.querySelector('.lg:col-span-2 .space-y-3');
    if (!container) return;
    
    container.innerHTML = '';
    
    reservations.slice(0, 5).forEach(r => {
        const isCheckin = new Date(r.check_in).toDateString() === new Date().toDateString();
        const bgClass = isCheckin ? 'bg-green-50' : 'bg-blue-50';
        const iconClass = isCheckin ? 'text-green-600' : 'text-blue-600';
        const icon = isCheckin ? 'fa-sign-in-alt' : 'fa-sign-out-alt';
        
        container.innerHTML += `
            <div class="flex items-center justify-between p-3 ${bgClass} rounded-lg cursor-pointer hover:shadow-md" 
                 onclick="getReservationDetails(${r.id})">
                <div class="flex items-center space-x-3">
                    <div class="w-10 h-10 ${bgClass.replace('50', '200')} rounded-full flex items-center justify-center">
                        <i class="fas ${icon} ${iconClass}"></i>
                    </div>
                    <div>
                        <p class="font-medium">${r.guest?.full_name || 'Гість'}</p>
                        <p class="text-sm text-gray-500">Номер ${r.room?.number || 'не призначено'} • ${r.adults} дорослих</p>
                    </div>
                </div>
                <div class="text-right">
                    <p class="text-sm font-medium ${isCheckin ? 'text-green-600' : 'text-blue-600'}">
                        ${isCheckin ? 'Заїзд' : 'Виїзд'} ${new Date(isCheckin ? r.check_in : r.check_out).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'})}
                    </p>
                    <p class="text-xs text-gray-400">${r.total_amount} ₴</p>
                </div>
            </div>
        `;
    });
}
