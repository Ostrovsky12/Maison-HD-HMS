"""
OTA Integration - синхронізація з Booking.com, Expedia та іншими
"""

import requests
import json
from datetime import datetime, timedelta
from loguru import logger
from app import db
from models.ota_channel import OTAChannel
from models.reservation import Reservation
from models.room import Room, RoomCategory
from services.telegram_bot import send_notification


class OTAIntegrationService:
    """Сервіс для інтеграції з OTA каналами"""
    
    def __init__(self, hotel_id):
        self.hotel_id = hotel_id
        self.channels = OTAChannel.query.filter_by(
            hotel_id=hotel_id,
            sync_enabled=True
        ).all()
    
    def sync_all_channels(self):
        """Синхронізація з усіма активними каналами"""
        results = []
        
        for channel in self.channels:
            try:
                result = self.sync_channel(channel)
                results.append({
                    'channel': channel.name,
                    'status': 'success',
                    'bookings': result
                })
            except Exception as e:
                logger.error(f"OTA sync failed for {channel.name}: {e}")
                results.append({
                    'channel': channel.name,
                    'status': 'error',
                    'error': str(e)
                })
        
        return results
    
    def sync_channel(self, channel):
        """Синхронізація з конкретним каналом"""
        logger.info(f"Syncing with {channel.name}")
        
        # Отримуємо нові бронювання з OTA
        new_bookings = self._fetch_new_bookings(channel)
        
        # Імпортуємо бронювання в систему
        imported = []
        for booking_data in new_bookings:
            try:
                reservation = self._import_booking(channel, booking_data)
                imported.append(reservation.reservation_number)
            except Exception as e:
                logger.error(f"Failed to import booking from {channel.name}: {e}")
        
        # Оновлюємо доступність в OTA
        self._update_availability(channel)
        
        # Оновлюємо час останньої синхронізації
        channel.last_sync = datetime.utcnow()
        channel.sync_status = 'success'
        db.session.commit()
        
        send_notification(
            f"🔄 Синхронізація з {channel.name} завершена\n"
            f"Імпортовано бронювань: {len(imported)}"
        )
        
        return imported
    
    def _fetch_new_bookings(self, channel):
        """Отримання нових бронювань з OTA API"""
        # TODO: Реалізація залежить від конкретного OTA
        # Тут приклад для Booking.com
        if channel.channel_type == 'booking':
            return self._fetch_booking_bookings(channel)
        elif channel.channel_type == 'expedia':
            return self._fetch_expedia_bookings(channel)
        return []
    
    def _fetch_booking_bookings(self, channel):
        """Отримання бронювань з Booking.com"""
        # Приклад API запиту до Booking.com
        headers = {
            'Authorization': f'Bearer {channel.api_key}',
            'Content-Type': 'application/json'
        }
        
        params = {
            'last_update': channel.last_sync.isoformat() if channel.last_sync else None
        }
        
        try:
            response = requests.get(
                f"{channel.api_url}/bookings",
                headers=headers,
                params=params,
                timeout=30
            )
            
            if response.status_code == 200:
                return response.json().get('bookings', [])
            else:
                logger.error(f"Booking.com API error: {response.status_code}")
                return []
                
        except Exception as e:
            logger.error(f"Failed to fetch from Booking.com: {e}")
            return []
    
    def _import_booking(self, channel, booking_data):
        """Імпорт бронювання з OTA в систему"""
        from models.guest import Guest
        
        # Пошук або створення гостя
        guest = Guest.query.filter_by(
            email=booking_data.get('guest_email')
        ).first()
        
        if not guest:
            guest = Guest(
                first_name=booking_data['guest_first_name'],
                last_name=booking_data['guest_last_name'],
                email=booking_data.get('guest_email'),
                phone=booking_data.get('guest_phone')
            )
            db.session.add(guest)
            db.session.commit()
        
        # Пошук категорії номера
        category = RoomCategory.query.filter_by(
            hotel_id=self.hotel_id,
            code=booking_data.get('room_category')
        ).first()
        
        if not category:
            raise ValueError(f"Room category not found: {booking_data.get('room_category')}")
        
        # Створення бронювання
        check_in = datetime.fromisoformat(booking_data['check_in'])
        check_out = datetime.fromisoformat(booking_data['check_out'])
        
        reservation = Reservation(
            reservation_number=Reservation.generate_number(),
            guest_id=guest.id,
            check_in=check_in,
            check_out=check_out,
            room_category_id=category.id,
            adults=booking_data.get('adults', 1),
            children=booking_data.get('children', 0),
            total_amount=booking_data.get('total_amount', 0),
            source=channel.channel_type,
            source_id=booking_data.get('booking_id'),
            notes=f"Імпортовано з {channel.name}"
        )
        
        db.session.add(reservation)
        db.session.commit()
        
        # Оновлення статистики каналу
        channel.total_bookings += 1
        channel.total_revenue += reservation.total_amount
        db.session.commit()
        
        return reservation
    
    def _update_availability(self, channel):
        """Оновлення доступності номерів в OTA"""
        # Отримуємо доступність на найближчі 90 днів
        availability = {}
        start_date = datetime.now().date()
        
        for i in range(90):
            date = start_date + timedelta(days=i)
            availability[date.isoformat()] = {}
            
            for category in RoomCategory.query.filter_by(hotel_id=self.hotel_id, is_active=True):

                available = total_rooms - occupied
                
                availability[date.isoformat()][category.code] = {
                    'available': available,
                    'price': category.base_price
                }
        
        # Відправка доступності в OTA
        if channel.channel_type == 'booking':
            self._send_booking_availability(channel, availability)
        elif channel.channel_type == 'expedia':
            self._send_expedia_availability(channel, availability)
    
    def _get_occupied_count(self, category_id, date):

class LockIntegrationService:
    """Сервіс для інтеграції з системами замків"""
    
    def __init__(self, hotel_id):
        self.hotel_id = hotel_id
        self.config = IntegrationConfig.query.filter_by(
            hotel_id=hotel_id,
            integration_type='lock',
            is_active=True
        ).first()
    
    def create_key_card(self, reservation):
        """Створення ключ-картки для бронювання"""
        if not self.config:
            logger.warning("No lock integration configured")
            return None
        
        lock_type = self.config.config.get('lock_type')
        
        if lock_type == 'vingcard':
            return self._create_vingcard_key(reservation)
        elif lock_type == 'salto':
            return self._create_salto_key(reservation)
        else:
            logger.error(f"Unknown lock type: {lock_type}")
            return None
    
    def _create_vingcard_key(self, reservation):
        """Створення ключа для VingCard"""
        config = self.config.config
        
        # Дані для ключа
        key_data = {
            'room_number': reservation.room.number if reservation.room else None,
            'check_in': reservation.check_in.isoformat(),
            'check_out': reservation.check_out.isoformat(),
            'guest_name': reservation.guest.full_name,
            'reservation_id': reservation.id
        }
        
        # Підписання запиту
        signature = self._generate_signature(key_data, config.get('api_secret'))
        
        headers = {
            'Authorization': f'Bearer {config.get("api_key")}',
            'X-Signature': signature,
            'Content-Type': 'application/json'
        }
        
        try:
            response = requests.post(
                f"{config.get('api_url')}/keys/create",
                json=key_data,
                headers=headers,
                timeout=30
            )
            
            if response.status_code == 200:
                result = response.json()
                logger.info(f"VingCard key created for reservation {reservation.id}")
                return result.get('key_code')
            else:
                logger.error(f"VingCard API error: {response.status_code}")
                return None
                
        except Exception as e:
            logger.error(f"Failed to create VingCard key: {e}")
            return None

class ForecastingService:
    """Сервіс прогнозування"""
    
    def __init__(self, hotel_id):
        self.hotel_id = hotel_id
    
    def generate_forecast(self, days=90):
        """Генерація прогнозу на N днів вперед"""
        # Отримуємо історичні дані
        history = self._get_history()
        
        if len(history) < 30:
            logger.warning("Not enough history for forecasting")
            return []
        
        # Генеруємо прогнози
        forecasts = []
        start_date = datetime.now().date()
        
        for i in range(days):
            date = start_date + timedelta(days=i)
            
            # Прогноз заповненості
            occupancy = self._predict_occupancy(history, date)
            
            # Прогноз ADR
            adr = self._predict_adr(history, date)
            
            # Прогноз доходу
            total_rooms = self._get_total_rooms()
            revenue = occupancy / 100 * total_rooms * adr
            
            # Довірчі інтервали
            confidence = self._calculate_confidence(history, occupancy)
            
            forecast = Forecast(
                hotel_id=self.hotel_id,
                forecast_date=date,
                predicted_occupancy=occupancy,
                predicted_revenue=revenue,
                predicted_adr=adr,
                confidence_lower=confidence['lower'],
                confidence_upper=confidence['upper'],
                season_factor=self._get_season_factor(date),
                events_factor=self._get_events_factor(date),
                competitor_factor=self._get_competitor_factor(date)
            )
            
            forecasts.append(forecast)
        
        # Зберігаємо прогнози
        db.session.add_all(forecasts)
        db.session.commit()
        
        logger.info(f"Generated {days} days forecast for hotel {self.hotel_id}")
        
        return forecasts
    
    def _get_history(self):
        """Отримання історичних даних"""
        history = KPIHistory.query.filter_by(
            hotel_id=self.hotel_id
        ).order_by(KPIHistory.date.desc()).limit(365).all()
        
        return [{
            'date': h.date,
            'occupancy': h.occupancy_rate,
            'adr': h.adr,
            'revenue': h.total_revenue,
            'weekday': h.date.weekday(),
            'month': h.date.month
        } for h in history]
    
    def _predict_occupancy(self, history, date):
        """Прогноз заповненості на дату"""
        # Використовуємо поліноміальну регресію
        X = []
        y = []
        
        for h in history:
            X.append([
                h['date'].weekday(),      # день тижня
                h['date'].month,           # місяць
                h['date'].day,             # день місяця
                self._is_holiday(h['date']),  # чи свято
                self._is_weekend(h['date'])   # чи вихідний
            ])
            y.append(h['occupancy'])
        
        if len(X) < 10:
            return 50  # default
        
        # Поліноміальна регресія
        poly = PolynomialFeatures(degree=2)
        X_poly = poly.fit_transform(X)
        
        model = LinearRegression()
        model.fit(X_poly, y)
        
        # Прогноз для нової дати
        X_new = [[
            date.weekday(),
            date.month,
            date.day,
            self._is_holiday(date),
            self._is_weekend(date)
        ]]
        X_new_poly = poly.transform(X_new)
        
        prediction = model.predict(X_new_poly)[0]
        
        # Обмежуємо значення 0-100
        return max(0, min(100, prediction))
    
    def _predict_adr(self, history, date):
        """Прогноз середньої ціни номеру"""
        # ADR = Average Daily Rate
        if not history:
            return 1500  # default
        
        # Проста середня за останні 30 днів з сезонним коефіцієнтом
        recent = [h['adr'] for h in history[:30] if h['adr'] > 0]
        avg_adr = sum(recent) / len(recent) if recent else 1500
        
        season_factor = self._get_season_factor(date)
        weekday_factor = 1.2 if date.weekday() >= 4 else 1.0
        
        return avg_adr * season_factor * weekday_factor
    
    def _calculate_confidence(self, history, prediction):
        """Розрахунок довірчого інтервалу"""
        # Використовуємо стандартне відхилення історичних даних
        occupancies = [h['occupancy'] for h in history[:30]]
        
        if len(occupancies) < 2:
            return {'lower': prediction - 10, 'upper': prediction + 10}
        
        std_dev = np.std(occupancies)
        confidence_interval = std_dev * 1.96  # 95% довірчий інтервал
        
        return {
            'lower': max(0, prediction - confidence_interval),
            'upper': min(100, prediction + confidence_interval)
        }
    
    def _get_season_factor(self, date):
        """Отримання сезонного коефіцієнта"""
        month = date.month
        
        # Високий сезон (червень-серпень)
        if month in [6, 7, 8]:
            return 1.3
        # Середній сезон (травень, вересень)
        elif month in [5, 9]:
            return 1.1
        # Низький сезон
        elif month in [11, 12, 1, 2, 3]:
            return 0.85
        else:
            return 1.0
    
    def _get_events_factor(self, date):
        """Фактор наявності подій поруч"""
        # TODO: Інтеграція з календарем подій
        return 1.0

        """Фактор цін конкурентів"""
        # TODO: Інтеграція з конкурентним аналізом
        return 1.0
    
    def _is_weekend(self, date):
        """Перевірка чи вихідний день"""
        return 1 if date.weekday() >= 5 else 0
    
    def _is_holiday(self, date):
        """Перевірка чи святковий день"""
        holidays = ['01-01', '01-07', '03-08', '05-01', '05-09', '06-28', '08-24']
        return 1 if date.strftime('%m-%d') in holidays else 0

        """Відправка email через SMTP"""
        if not self.smtp_server:
            logger.warning("SMTP not configured")
            return False
        
        try:
            msg = MIMEMultipart('alternative')
            msg['Subject'] = subject
            msg['From'] = self.from_email
            msg['To'] = to_email
            
            part = MIMEText(html_content, 'html')
            msg.attach(part)
            
            server = smtplib.SMTP(self.smtp_server, self.smtp_port)
            server.starttls()
            server.login(self.smtp_user, self.smtp_password)
            server.sendmail(self.from_email, to_email, msg.as_string())
            server.quit()
            
            logger.info(f"Email sent to {to_email}: {subject}")
            return True
            
        except Exception as e:
            logger.error(f"Failed to send email: {e}")
            return False

    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Python
      uses: actions/setup-python@v4

        python-version: '3.11'

        pip install pytest pytest-cov
    
    - name: Run tests
      env:
        DATABASE_URL: postgresql://test_user:test_pass@localhost:5432/test_db
        FLASK_ENV: testing
      run: |

    - name: Upload coverage
      uses: codecov/codecov-action@v3
      with:
        file: ./coverage.xml

    needs: test
    runs-on: ubuntu-latest

  deploy:

    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'

# Maison HD - Hotel Management System

Сучасна система управління готелем (PMS) з відкритим API, розширеною аналітикою та підтримкою мобільних пристроїв.

## 🚀 Основні можливості

- **Управління бронюваннями** - шахматка, групові бронювання, онлайн-бронювання
- **CRM для гостей** - історія відвідувань, програми лояльності, компанії
- **Служба покоївок** - мобільний додаток, контроль якості, автоматичне призначення
- **Фінанси** - рахунки, нічний аудит, інтеграція з 1С
- **Інтеграції** - OTA (Booking.com, Expedia), системи замків (VingCard), Telegram
- **Аналітика** - прогнозування заповненості, KPI дашборди, звіти
- **PWA** - мобільний інтерфейс для покоївок, push-сповіщення

## 📦 Технології

- **Backend**: Python, Flask, SQLAlchemy, Celery
- **Database**: PostgreSQL, Redis
- **Frontend**: TailwindCSS, Chart.js, Vanilla JS
- **Infrastructure**: Docker, Nginx, GitHub Actions
- **Integrations**: REST API, Webhooks, Telegram Bot
