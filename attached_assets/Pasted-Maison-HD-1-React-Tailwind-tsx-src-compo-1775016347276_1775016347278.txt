// src/components/booking/Chessboard.tsx

import { useState, useRef, useEffect, useCallback } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { api } from '@/lib/api';
import { toast } from 'sonner';
import {
  ChevronLeft,
  ChevronRight,
  Calendar,
  Filter,
  Download,
  Settings,
  Plus,
  Bed,
  Users,
  Clock,
  Eye,
  MoreVertical,
} from 'lucide-react';

interface Room {
  id: number;
  number: string;
  name: string;
  category: string;
  capacity: number;
  floor: number;
  status: 'clean' | 'dirty' | 'repair' | 'occupied';
}

interface Booking {
  id: number;
  roomId: number;
  guestName: string;
  checkIn: string;
  checkOut: string;
  status: 'confirmed' | 'checked_in' | 'checked_out' | 'cancelled';
  adults: number;
  children: number;
  phone: string;
  totalAmount: number;
  notes?: string;
}

interface ChessboardProps {
  propertyId: number;
  startDate?: Date;
}

export function Chessboard({ propertyId, startDate = new Date() }: ChessboardProps) {
  // Стан
  const [currentDate, setCurrentDate] = useState(startDate);
  const [viewMode, setViewMode] = useState<'day' | 'week' | 'month'>('month');
  const [selectedRooms, setSelectedRooms] = useState<number[]>([]);
  const [selectionRange, setSelectionRange] = useState<{ start: string | null; end: string | null }>({ start: null, end: null });
  const [isSelecting, setIsSelecting] = useState(false);
  const [tooltip, setTooltip] = useState<{ x: number; y: number; text: string; visible: boolean }>({ x: 0, y: 0, text: '', visible: false });
  const [showCreateBooking, setShowCreateBooking] = useState(false);
  const [hoveredCell, setHoveredCell] = useState<{ roomId: number; date: string } | null>(null);
  
  const chessboardRef = useRef<HTMLDivElement>(null);
  const cellRefs = useRef<Map<string, HTMLDivElement>>(new Map());
  
  // Отримання даних
  const { data: rooms, isLoading: roomsLoading } = useQuery({
    queryKey: ['rooms', propertyId],
    queryFn: () => api.get(`/properties/${propertyId}/rooms`),
  });
  
  const { data: bookings, refetch: refetchBookings } = useQuery({
    queryKey: ['bookings', propertyId, currentDate, viewMode],
    queryFn: () => api.get(`/properties/${propertyId}/bookings/calendar`, {
      params: {
        startDate: currentDate.toISOString().split('T')[0],
        days: viewMode === 'month' ? 35 : viewMode === 'week' ? 7 : 1,
      },
    }),
  });
  
  // Мутація для створення бронювання
  const createBookingMutation = useMutation({
    mutationFn: (data: any) => api.post('/bookings', data),
    onSuccess: () => {
      toast.success('Бронювання створено!');
      refetchBookings();
      setShowCreateBooking(false);
      setSelectionRange({ start: null, end: null });
    },
    onError: () => toast.error('Помилка створення бронювання'),
  });
  
  // Генерація днів для відображення
  const generateDays = useCallback(() => {
    const days = [];
    const start = new Date(currentDate);
    
    if (viewMode === 'month') {
      start.setDate(1);
      const firstDayOfMonth = start.getDay() || 7;
      start.setDate(start.getDate() - (firstDayOfMonth - 1));
      
      for (let i = 0; i < 35; i++) {
        const date = new Date(start);
        date.setDate(start.getDate() + i);
        days.push(date);
      }
    } else if (viewMode === 'week') {
      const dayOfWeek = start.getDay() || 7;
      start.setDate(start.getDate() - (dayOfWeek - 1));
      
      for (let i = 0; i < 7; i++) {
        const date = new Date(start);
        date.setDate(start.getDate() + i);
        days.push(date);
      }
    } else {
      days.push(new Date(currentDate));
    }
    
    return days;
  }, [currentDate, viewMode]);
  
  const days = generateDays();
  
  // Отримання бронювання для клітинки
  const getBookingForCell = (roomId: number, date: Date): Booking | null => {
    if (!bookings?.data) return null;
    const dateStr = date.toISOString().split('T')[0];
    return bookings.data.find(
      (b: Booking) => b.roomId === roomId && b.checkIn <= dateStr && b.checkOut > dateStr
    ) || null;
  };
  
  // Визначення статусу клітинки
  const getCellStatus = (room: Room, date: Date, booking: Booking | null) => {
    if (booking) {
      if (booking.status === 'checked_in') return 'occupied';
      if (booking.status === 'confirmed') return 'booked';
      return 'blocked';
    }
    
    if (room.status === 'dirty') return 'dirty';
    if (room.status === 'repair') return 'repair';
    return 'free';
  };
  
  // Перевірка чи дата в діапазоні виділення
  const isInSelectionRange = (roomId: number, date: Date): boolean => {
    if (!selectionRange.start || !selectionRange.end) return false;
    const dateStr = date.toISOString().split('T')[0];
    return dateStr >= selectionRange.start && dateStr <= selectionRange.end;
  };
  
  // Обробка виділення мишею
  const handleMouseDown = (roomId: number, date: Date) => {
    setIsSelecting(true);
    setSelectionRange({
      start: date.toISOString().split('T')[0],
      end: date.toISOString().split('T')[0],
    });
    setSelectedRooms([roomId]);
  };
  
  const handleMouseEnter = (roomId: number, date: Date) => {
    if (!isSelecting) return;
    
    setSelectionRange(prev => ({
      ...prev,
      end: date.toISOString().split('T')[0],
    }));
    
    if (!selectedRooms.includes(roomId)) {
      setSelectedRooms(prev => [...prev, roomId]);
    }
  };
  
  const handleMouseUp = () => {
    if (isSelecting && selectionRange.start && selectionRange.end) {
      setShowCreateBooking(true);
    }
    setIsSelecting(false);
  };
  
  // Підказки при наведенні
  const showTooltip = (event: React.MouseEvent, text: string) => {
    setTooltip({
      x: event.clientX + 15,
      y: event.clientY - 30,
      text,
      visible: true,
    });
  };
  
  const hideTooltip = () => {
    setTooltip({ ...tooltip, visible: false });
  };
  
  // Навігація
  const navigate = (direction: 'prev' | 'next') => {
    const newDate = new Date(currentDate);
    if (viewMode === 'month') {
      newDate.setMonth(newDate.getMonth() + (direction === 'next' ? 1 : -1));
    } else if (viewMode === 'week') {
      newDate.setDate(newDate.getDate() + (direction === 'next' ? 7 : -7));
    } else {
      newDate.setDate(newDate.getDate() + (direction === 'next' ? 1 : -1));
    }
    setCurrentDate(newDate);
  };
  
  // Стилі для статусів клітинок
  const cellStyles = {
    free: 'bg-green-50 hover:bg-green-100 border-green-200',
    occupied: 'bg-red-100 hover:bg-red-200 border-red-200 cursor-not-allowed',
    booked: 'bg-yellow-100 hover:bg-yellow-200 border-yellow-200',
    dirty: 'bg-orange-100 hover:bg-orange-200 border-orange-200',
    repair: 'bg-gray-200 hover:bg-gray-300 border-gray-300 cursor-not-allowed',
    blocked: 'bg-purple-100 hover:bg-purple-200 border-purple-200',
  };
  
  if (roomsLoading) {
    return (
      <div className="flex items-center justify-center h-96">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-indigo-600"></div>
      </div>
    );
  }
  
  return (
    <div className="bg-white rounded-xl shadow-lg overflow-hidden">
      {/* Tooltip */}
      {tooltip.visible && (
        <div
          className="fixed z-50 bg-gray-900 text-white text-sm px-3 py-1.5 rounded-lg shadow-lg pointer-events-none whitespace-nowrap"
          style={{ left: tooltip.x, top: tooltip.y }}
        >
          {tooltip.text}
        </div>
      )}
      
      {/* Header */}
      <div className="border-b border-gray-200 bg-gradient-to-r from-gray-50 to-white px-6 py-4">
        <div className="flex flex-wrap items-center justify-between gap-4">
          <div className="flex items-center space-x-4">
            <button
              onClick={() => navigate('prev')}
              className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
            >
              <ChevronLeft className="w-5 h-5 text-gray-600" />
            </button>
            
            <h2 className="text-xl font-bold text-gray-800">
              {currentDate.toLocaleString('uk-UA', { month: 'long', year: 'numeric' })}
            </h2>
            
            <button
              onClick={() => navigate('next')}
              className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
            >
              <ChevronRight className="w-5 h-5 text-gray-600" />
            </button>
            
            <button
              onClick={() => setCurrentDate(new Date())}
              className="px-3 py-1.5 text-sm bg-indigo-50 text-indigo-600 rounded-lg hover:bg-indigo-100 transition-colors"
            >
              Сьогодні
            </button>
          </div>
          
          <div className="flex items-center space-x-3">
            <div className="flex bg-gray-100 rounded-lg p-1">
              {(['day', 'week', 'month'] as const).map((mode) => (
                <button
                  key={mode}
                  onClick={() => setViewMode(mode)}
                  className={`px-4 py-1.5 text-sm rounded-md transition-all ${
                    viewMode === mode
                      ? 'bg-white text-indigo-600 shadow-sm'
                      : 'text-gray-600 hover:bg-gray-200'
                  }`}
                >
                  {mode === 'day' ? 'День' : mode === 'week' ? 'Тиждень' : 'Місяць'}
                </button>
              ))}
            </div>
            
            <button className="p-2 hover:bg-gray-100 rounded-lg transition-colors">
              <Filter className="w-5 h-5 text-gray-500" />
            </button>
            
            <button className="p-2 hover:bg-gray-100 rounded-lg transition-colors">
              <Download className="w-5 h-5 text-gray-500" />
            </button>
            
            <button className="p-2 hover:bg-gray-100 rounded-lg transition-colors">
              <Settings className="w-5 h-5 text-gray-500" />
            </button>
          </div>
        </div>
        
        {/* Легенда */}
        <div className="flex flex-wrap gap-4 mt-4 text-xs">
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-green-100 border border-green-200 rounded"></div>
            <span className="text-gray-600">Вільний</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-yellow-100 border border-yellow-200 rounded"></div>
            <span className="text-gray-600">Заброньовано</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-red-100 border border-red-200 rounded"></div>
            <span className="text-gray-600">Заселено</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-orange-100 border border-orange-200 rounded"></div>
            <span className="text-gray-600">Потребує прибирання</span>
          </div>
          <div className="flex items-center space-x-1.5">
            <div className="w-3 h-3 bg-gray-200 border border-gray-300 rounded"></div>
            <span className="text-gray-600">На ремонті</span>
          </div>
          <div className="flex items-center space-x-1.5 ml-4">
            <div className="w-3 h-3 bg-indigo-200 border border-indigo-300 rounded"></div>
            <span className="text-gray-600">Виділено для бронювання</span>
          </div>
        </div>
      </div>
      
      {/* Шахівка */}
      <div
        ref={chessboardRef}
        className="overflow-x-auto"
        onMouseUp={handleMouseUp}
      >
        <div className="min-w-[800px]">
          {/* Заголовок з датами */}
          <div className="flex border-b border-gray-200 bg-gray-50">
            <div className="w-32 flex-shrink-0 p-3 text-left text-sm font-medium text-gray-500 border-r border-gray-200">
              Номер
            </div>
            <div className="flex-1 grid" style={{ gridTemplateColumns: `repeat(${days.length}, minmax(100px, 1fr))` }}>
              {days.map((date, idx) => (
                <div
                  key={idx}
                  className={`p-2 text-center border-r border-gray-200 ${
                    date.toDateString() === new Date().toDateString() ? 'bg-indigo-50' : ''
                  }`}
                >
                  <div className="text-sm font-medium text-gray-700">
                    {date.toLocaleString('uk-UA', { day: 'numeric', month: 'short' })}
                  </div>
                  <div className="text-xs text-gray-400">
                    {date.toLocaleString('uk-UA', { weekday: 'short' })}
                  </div>
                </div>
              ))}
            </div>
          </div>
          
          {/* Рядки з номерами */}
          {rooms?.data?.map((room: Room) => (
            <div key={room.id} className="flex border-b border-gray-100 hover:bg-gray-50 transition-colors">
              {/* Інформація про номер */}
              <div className="w-32 flex-shrink-0 p-3 border-r border-gray-200">
                <div className="font-medium text-gray-800">{room.number}</div>
                <div className="text-xs text-gray-400 flex items-center space-x-1 mt-0.5">
                  <Bed className="w-3 h-3" />
                  <span>{room.capacity}</span>
                  <span className="mx-1">•</span>
                  <span>{room.category}</span>
                </div>
                <div className="text-xs text-gray-400 mt-1">
                  {room.name}
                </div>
              </div>
              
              {/* Клітинки днів */}
              <div
                className="flex-1 grid"
                style={{ gridTemplateColumns: `repeat(${days.length}, minmax(100px, 1fr))` }}
              >
                {days.map((date, idx) => {
                  const booking = getBookingForCell(room.id, date);
                  const status = getCellStatus(room, date, booking);
                  const isSelected = isInSelectionRange(room.id, date);
                  const isHovered = hoveredCell?.roomId === room.id && hoveredCell?.date === date.toISOString().split('T')[0];
                  
                  const statusStyle = cellStyles[status as keyof typeof cellStyles];
                  const isInteractive = status === 'free' || status === 'dirty';
                  
                  return (
                    <div
                      key={idx}
                      ref={(el) => {
                        if (el) {
                          cellRefs.current.set(`${room.id}-${date.toISOString().split('T')[0]}`, el);
                        }
                      }}
                      className={`
                        relative p-2 border-r border-b border-gray-100 transition-all duration-150
                        ${statusStyle}
                        ${isSelected ? 'ring-2 ring-indigo-400 ring-inset bg-indigo-100' : ''}
                        ${isHovered ? 'scale-[0.98] shadow-inner' : ''}
                        ${isInteractive ? 'cursor-pointer' : 'cursor-not-allowed opacity-70'}
                      `}
                      onMouseDown={() => isInteractive && handleMouseDown(room.id, date)}
                      onMouseEnter={(e) => {
                        if (isInteractive) {
                          handleMouseEnter(room.id, date);
                          setHoveredCell({ roomId: room.id, date: date.toISOString().split('T')[0] });
                        }
                        // Підказка
                        if (booking) {
                          showTooltip(e, `${booking.guestName}\n📅 ${booking.checkIn} → ${booking.checkOut}\n👥 ${booking.adults} дорослих, ${booking.children} дітей\n💰 ${booking.totalAmount} ₴`);
                        } else {
                          showTooltip(e, `Номер ${room.number}\n✨ Вільний\n🖱️ Натисніть та протягніть для бронювання`);
                        }
                      }}
                      onMouseLeave={() => {
                        hideTooltip();
                        setHoveredCell(null);
                      }}
                    >
                      {booking ? (
                        <div className="text-xs">
                          <div className="font-medium text-gray-800 truncate">
                            {booking.guestName}
                          </div>
                          <div className="text-gray-500 text-[10px] mt-0.5">
                            {booking.adults} дор. • {booking.children} діт.
                          </div>
                          {booking.status === 'checked_in' && (
                            <div className="mt-1 text-[10px] text-red-500 font-medium">
                              Заселено
                            </div>
                          )}
                        </div>
                      ) : (
                        <div className="text-center text-gray-400 text-[10px]">
                          {status === 'free' && '✓ Вільний'}
                          {status === 'dirty' && '🧹 Потребує прибирання'}
                          {status === 'repair' && '🔧 На ремонті'}
                        </div>
                      )}
                      
                      {/* Індикатор діапазону */}
                      {isSelected && selectionRange.start && selectionRange.end && (
                        <div className="absolute inset-0 bg-indigo-100 opacity-30 pointer-events-none" />
                      )}
                    </div>
                  );
                })}
              </div>
            </div>
          ))}
        </div>
      </div>
      
      {/* Модальне вікно створення бронювання */}
      {showCreateBooking && selectionRange.start && selectionRange.end && (
        <CreateBookingModal
          propertyId={propertyId}
          checkIn={selectionRange.start}
          checkOut={selectionRange.end}
          roomIds={selectedRooms}
          onClose={() => {
            setShowCreateBooking(false);
            setSelectionRange({ start: null, end: null });
            setSelectedRooms([]);
          }}
          onSubmit={(data) => createBookingMutation.mutate(data)}
          isLoading={createBookingMutation.isPending}
        />
      )}
    </div>
  );
}

// src/components/booking/CreateBookingModal.tsx

import { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { api } from '@/lib/api';
import { X, Search, User, Calendar, Users, DollarSign, FileText, ChevronDown } from 'lucide-react';

interface CreateBookingModalProps {
  propertyId: number;
  checkIn: string;
  checkOut: string;
  roomIds: number[];
  onClose: () => void;
  onSubmit: (data: any) => void;
  isLoading: boolean;
}

export function CreateBookingModal({
  propertyId,
  checkIn,
  checkOut,
  roomIds,
  onClose,
  onSubmit,
  isLoading,
}: CreateBookingModalProps) {
  const [step, setStep] = useState(1);
  const [guestSearch, setGuestSearch] = useState('');
  const [selectedGuest, setSelectedGuest] = useState<any>(null);
  const [formData, setFormData] = useState({
    guestId: null as number | null,
    adults: 2,
    children: 0,
    extraBeds: 0,
    specialRequests: '',
    notes: '',
    source: 'direct',
  });
  
  // Пошук гостей
  const { data: guests, isLoading: guestsLoading } = useQuery({
    queryKey: ['guests-search', guestSearch],
    queryFn: () => api.get(`/guests/search?q=${guestSearch}`),
    enabled: guestSearch.length > 2,
  });
  
  // Розрахунок вартості
  const { data: priceData } = useQuery({
    queryKey: ['booking-price', propertyId, checkIn, checkOut, roomIds[0], formData.adults, formData.extraBeds],
    queryFn: () => api.get(`/bookings/price`, {
      params: {
        propertyId,
        checkIn,
        checkOut,
        roomId: roomIds[0],
        adults: formData.adults,
        extraBeds: formData.extraBeds,
      },
    }),
    enabled: roomIds.length > 0,
  });
  
  const nights = Math.ceil((new Date(checkOut).getTime() - new Date(checkIn).getTime()) / (1000 * 3600 * 24));
  const totalAmount = priceData?.data?.total || 0;
  
  const handleSubmit = () => {
    onSubmit({
      ...formData,
      guestId: selectedGuest?.id,
      propertyId,
      checkIn,
      checkOut,
      roomIds,
      totalAmount,
      nights,
    });
  };
  
  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-2xl max-w-2xl w-full max-h-[90vh] overflow-y-auto shadow-2xl">
        {/* Header */}
        <div className="sticky top-0 bg-white border-b border-gray-200 px-6 py-4 flex justify-between items-center">
          <div>
            <h2 className="text-xl font-bold text-gray-800">Нове бронювання</h2>
            <div className="text-sm text-gray-500 mt-1">
              {new Date(checkIn).toLocaleDateString('uk-UA')} — {new Date(checkOut).toLocaleDateString('uk-UA')} • {nights} {nights === 1 ? 'ніч' : 'ночей'}
            </div>
          </div>
          <button onClick={onClose} className="p-2 hover:bg-gray-100 rounded-lg transition-colors">
            <X className="w-5 h-5 text-gray-500" />
          </button>
        </div>
        
        {/* Steps */}
        <div className="px-6 py-4 border-b border-gray-100">
          <div className="flex items-center space-x-4">
            {[1, 2, 3].map((s) => (
              <div key={s} className="flex items-center">
                <div className={`
                  w-8 h-8 rounded-full flex items-center justify-center text-sm font-medium
                  ${step >= s ? 'bg-indigo-600 text-white' : 'bg-gray-200 text-gray-500'}
                `}>
                  {s}
                </div>
                {s < 3 && <div className="w-12 h-0.5 bg-gray-200 mx-2" />}
              </div>
            ))}
          </div>
          <div className="flex justify-between mt-2 text-xs text-gray-500">
            <span>Гість</span>
            <span>Деталі</span>
            <span>Підтвердження</span>
          </div>
        </div>
        
        {/* Content */}
        <div className="p-6">
          {step === 1 && (
            <div className="space-y-6">
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  Пошук гостя
                </label>
                <div className="relative">
                  <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 w-4 h-4 text-gray-400" />
                  <input
                    type="text"
                    value={guestSearch}
                    onChange={(e) => setGuestSearch(e.target.value)}
                    placeholder="Введіть телефон, email або прізвище..."
                    className="w-full pl-10 pr-4 py-2 border border-gray-200 rounded-lg focus:ring-2 focus:ring-indigo-500 focus:border-transparent"
                  />
                </div>
                
                {guestsLoading && (
                  <div className="mt-4 text-center text-gray-500">Пошук...</div>
                )}
                
                {guests?.data?.length > 0 && (
                  <div className="mt-4 space-y-2 max-h-64 overflow-y-auto">
                    {guests.data.map((guest: any) => (
                      <div
                        key={guest.id}
                        onClick={() => {
                          setSelectedGuest(guest);
                          setFormData({ ...formData, guestId: guest.id });
                        }}
                        className={`
                          p-3 border rounded-lg cursor-pointer transition-all
                          ${selectedGuest?.id === guest.id
                            ? 'border-indigo-300 bg-indigo-50'
                            : 'border-gray-200 hover:border-indigo-200 hover:bg-gray-50'
                          }
                        `}
                      >
                        <div className="flex items-center justify-between">
                          <div>
                            <div className="font-medium text-gray-800">
                              {guest.lastName} {guest.firstName}
                            </div>
                            <div className="text-sm text-gray-500 mt-0.5">
                              {guest.phone} • {guest.email}
                            </div>
                          </div>
                          {selectedGuest?.id === guest.id && (
                            <div className="text-indigo-600">✓</div>
                          )}
                        </div>
                      </div>
                    ))}
                  </div>
                )}
              </div>
              
              <div className="flex justify-end pt-4">
                <button
                  onClick={() => setStep(2)}
                  disabled={!selectedGuest}
                  className="px-6 py-2 bg-indigo-600 text-white rounded-lg hover:bg-indigo-700 disabled:opacity-50 disabled:cursor-not-allowed transition-colors"
                >
                  Далі →
                </button>
              </div>
            </div>
          )}
          
          {step === 2 && (
            <div className="space-y-6">
              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Дорослі
                  </label>
                  <div className="flex items-center space-x-2">
                    <button
                      onClick={() => setFormData({ ...formData, adults: Math.max(1, formData.adults - 1) })}
                      className="w-8 h-8 border border-gray-200 rounded-lg hover:bg-gray-50"
                    >
                      -
                    </button>
                    <span className="w-12 text-center font-medium">{formData.adults}</span>
                    <button
                      onClick={() => setFormData({ ...formData, adults: Math.min(10, formData.adults + 1) })}
                      className="w-8 h-8 border border-gray-200 rounded-lg hover:bg-gray-50"
                    >
                      +
                    </button>
                  </div>
                </div>
                
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Діти
                  </label>
                  <div className="flex items-center space-x-2">
                    <button
                      onClick={() => setFormData({ ...formData, children: Math.max(0, formData.children - 1) })}
                      className="w-8 h-8 border border-gray-200 rounded-lg hover:bg-gray-50"
                    >
                      -
                    </button>
                    <span className="w-12 text-center font-medium">{formData.children}</span>
                    <button
                      onClick={() => setFormData({ ...formData, children: Math.min(5, formData.children + 1) })}
                      className="w-8 h-8 border border-gray-200 rounded-lg hover:bg-gray-50"
                    >
                      +
                    </button>
                  </div>

                    <span className="w-12 text-center font-medium">{formData.extraBeds}</span>
                    <button
                      onClick={() => setFormData({ ...formData, extraBeds: Math.min(2, formData.extraBeds + 1) })}
                      className="w-8 h-8 border border-gray-200 rounded-lg hover:bg-gray-50"

                      +
                    </button>
                  </div>
                </div>
                
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Джерело
                  </label>
                  <select
                    value={formData.source}
                    onChange={(e) => setFormData({ ...formData, source: e.target.value })}
                    className="w-full border border-gray-200 rounded-lg px-3 py-2"

                    <option value="direct">Пряме бронювання</option>
                    <option value="phone">По телефону</option>
                    <option value="email">Email</option>
                    <option value="booking">Booking.com</option>

                  className="w-full border border-gray-200 rounded-lg px-3 py-2 resize-none"
                />
              </div>
              
              <div className="flex justify-between pt-4">
                <button
                  onClick={() => setStep(1)}
                  className="px-6 py-2 border border-gray-200 rounded-lg hover:bg-gray-50 transition-colors"
                >
                  ← Назад
                </button>
                <button
                  onClick={() => setStep(3)}
                  className="px-6 py-2 bg-indigo-600 text-white rounded-lg hover:bg-indigo-700 transition-colors"
                >
                  Далі →
                </button>
              </div>
            </div>
          )}
          
          {step === 3 && (
            <div className="space-y-6">
              {/* Summary */}
              <div className="bg-gray-50 rounded-xl p-4 space-y-3">
                <div className="flex justify-between">
                  <span className="text-gray-600">Гість:</span>
                  <span className="font-medium text-gray-800">
                    {selectedGuest?.lastName} {selectedGuest?.firstName}
                  </span>
                </div>
                <div className="flex justify-between">
                  <span className="text-gray-600">Контакт:</span>
                  <span className="text-gray-800">{selectedGuest?.phone}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-gray-600">Дати:</span>
                  <span className="text-gray-800">
                    {new Date(checkIn).toLocaleDateString('uk-UA')} — {new Date(checkOut).toLocaleDateString('uk-UA')}
                  </span>
                </div>
                <div className="flex justify-between">
                  <span className="text-gray-600">Гості:</span>
                  <span className="text-gray-800">
                    {formData.adults} дор. {formData.children > 0 && `+ ${formData.children} діт.`}
                  </span>
                </div>
                <div className="border-t border-gray-200 pt-3 mt-2">
                  <div className="flex justify-between text-lg font-bold">
                    <span>До сплати:</span>
                    <span className="text-indigo-600">{totalAmount.toLocaleString()} ₴</span>
                  </div>
                  <div className="text-xs text-gray-500 text-right mt-1">
                    {nights} {nights === 1 ? 'ніч' : 'ночей'}
                  </div>
                </div>
              </div>
              
              <div className="bg-blue-50 rounded-xl p-4">
                <div className="flex items-start space-x-3">
                  <div className="w-8 h-8 bg-blue-100 rounded-full flex items-center justify-center flex-shrink-0">
                    <FileText className="w-4 h-4 text-blue-600" />
                  </div>
                  <div>
                    <p className="text-sm text-blue-800 font-medium">Підтвердження бронювання</p>
                    <p className="text-xs text-blue-600 mt-1">
                      Гість отримає SMS та Email з підтвердженням. Рахунок буде доступний для оплати онлайн.
                    </p>
                  </div>
                </div>
              </div>
              
              <div className="flex justify-between pt-4">
                <button
                  onClick={() => setStep(2)}
                  className="px-6 py-2 border border-gray-200 rounded-lg hover:bg-gray-50 transition-colors"
                >
                  ← Назад
                </button>
                <button
                  onClick={handleSubmit}
                  disabled={isLoading}
                  className="px-6 py-2 bg-indigo-600 text-white rounded-lg hover:bg-indigo-700 disabled:opacity-50 transition-colors"
                >
                  {isLoading ? 'Створення...' : 'Створити бронювання'}
                </button>
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

// server/routes/bookings.ts — додаємо ендпоінти для шахівки

// GET /api/properties/:id/bookings/calendar
router.get('/properties/:id/bookings/calendar', async (req, res) => {
  const propertyId = parseInt(req.params.id);
  const { startDate, days = 35 } = req.query;
  
  const start = new Date(startDate as string);
  const end = new Date(start);
  end.setDate(end.getDate() + parseInt(days as string));
  
  // Отримуємо всі бронювання на період
  const bookings = await db.select()
    .from(bookings)
    .where(
      and(
        eq(bookings.propertyId, propertyId),
        or(
          and(
            gte(bookings.checkIn, start),
            lt(bookings.checkIn, end)
          ),
          and(
            gte(bookings.checkOut, start),
            lt(bookings.checkOut, end)
          )
        )
      )
    )
    .orderBy(bookings.checkIn);
  
  // Отримуємо гостей для кожного бронювання
  const bookingsWithGuests = await Promise.all(
    bookings.map(async (booking) => {
      const guest = await db.select()
        .from(guests)
        .where(eq(guests.id, booking.guestId))
        .get();
      
      return {
        ...booking,
        guestName: guest ? `${guest.lastName} ${guest.firstName}` : 'Невідомий гість',
        guestPhone: guest?.phone,
      };
    })
  );
  
  res.json({ success: true, data: bookingsWithGuests });
});

// GET /api/bookings/price
router.get('/bookings/price', async (req, res) => {
  const { propertyId, checkIn, checkOut, roomId, adults, extraBeds } = req.query;
  
  const start = new Date(checkIn as string);
  const end = new Date(checkOut as string);
  const nights = Math.ceil((end.getTime() - start.getTime()) / (1000 * 3600 * 24));
  
  // Отримуємо ціну номера
  const room = await db.select()
    .from(rooms)
    .where(eq(rooms.id, parseInt(roomId as string)))
    .get();
  
  // Розрахунок з урахуванням сезонності
  let total = 0;
  const currentDate = new Date(start);
  
  for (let i = 0; i < nights; i++) {
    let price = room?.basePrice || 0;
    const month = currentDate.getMonth() + 1;
    
    // Сезонний коефіцієнт
    if (month >= 6 && month <= 8) price *= 1.3;
    else if (month >= 11 || month <= 3) price *= 0.85;
    
    // Вихідні
    if (currentDate.getDay() === 6 || currentDate.getDay() === 0) price *= 1.2;
    
    total += price;
    currentDate.setDate(currentDate.getDate() + 1);
  }
  
  // Додаткові місця
  const extraBedsCount = parseInt(extraBeds as string) || 0;
  total += extraBedsCount * 500;
  
  // Додаткові дорослі
  const adultsCount = parseInt(adults as string) || 2;
  if (adultsCount > 2) {
    total += (adultsCount - 2) * 300 * nights;
  }
  
  res.json({
    success: true,
    data: {
      nights,
      total,
      pricePerNight: total / nights,
    },
  });
});
